"use client"

import { useState, useEffect, Suspense } from "react"
import { useSearchParams, useRouter } from "next/navigation"
import { roomManager } from "@/lib/room-manager"
import { sessionManager } from "@/lib/supabase-session-manager"
import { resolveGameSessionId, STATS_BASE_URL } from "@/lib/game-session"

import { PageCornerLogos } from "@/components/layout/page-corner-logos"
import { PageTitle } from '@/components/layout/page-title'
import { sortPlayersForLeaderboard } from "@/lib/player-quiz-progress"
import { GameBackground } from "@/components/layout/game-background"
import { LeaderboardLoadingScreen, LeaderboardSimpleScreen } from "./components/leaderboard-screens"
import { FloatingActionButtons } from "./components/floating-action-buttons"
import { MobileLeaderboard } from "./components/mobile-leaderboard"
import { DesktopLeaderboard } from "./components/desktop-leaderboard"
import type { LeaderboardRoom } from "./components/types"

function LeaderboardHostPageContent() {
  const [room, setRoom] = useState<LeaderboardRoom | null>(null)
  const [loading, setLoading] = useState(true)
  const [hostId, setHostId] = useState<string | null>(null)
  const [isRestarting, setIsRestarting] = useState(false)
  const [showHomeTooltip, setShowHomeTooltip] = useState(false)
  const [showStatTooltip, setShowStatTooltip] = useState(false)
  const [showRestartTooltip, setShowRestartTooltip] = useState(false)
  const searchParams = useSearchParams()
  const router = useRouter()

  const roomCodeFromQuery = searchParams.get("roomCode")
  const [roomCode, setRoomCode] = useState<string | null>(roomCodeFromQuery)
  const [resolvingCode, setResolvingCode] = useState(true)
  const [sessionId, setSessionId] = useState<string | null>(null)

  useEffect(() => {
    const verifyHostAccess = async () => {
      if (!roomCodeFromQuery) {
        const loadFromSession = async () => {
          try {
            const sessionIdStored = sessionManager.getSessionIdFromStorage()
            if (sessionIdStored) {
              const sessionData = await sessionManager.getSessionData(sessionIdStored).catch(() => null)
              if (sessionData && sessionData.user_type === 'host') {
                const code = sessionData.user_data?.roomCode || sessionData.room_code
                if (code) {
                  setRoomCode(code)
                  setResolvingCode(false)
                  return
                }
              }
            }
          } catch { }

          try {
            const hostData = localStorage.getItem("currentHost")
            if (hostData) {
              const parsed = JSON.parse(hostData)
              if (parsed?.roomCode) {
                setRoomCode(parsed.roomCode)
                setResolvingCode(false)
                return
              }
            }
          } catch { }

          setResolvingCode(false)
          router.push("/")
        }
        loadFromSession()
        return
      }

      let verified = false

      try {
        const storedSessionId = sessionManager.getSessionIdFromStorage()
        if (storedSessionId) {
          const sessionData = await sessionManager.getSessionData(storedSessionId).catch(() => null)
          if (sessionData &&
            sessionData.user_type === 'host' &&
            (sessionData.user_data?.roomCode === roomCodeFromQuery || sessionData.room_code === roomCodeFromQuery)) {
            verified = true
          }
        }

        if (!verified && typeof window !== 'undefined') {
          const hostData = localStorage.getItem("currentHost")
          if (hostData) {
            const parsed = JSON.parse(hostData)
            if (parsed?.roomCode === roomCodeFromQuery) {
              verified = true

              try {
                console.log("[Leaderboard] Looking for session with game_pin:", roomCodeFromQuery)
                const resolvedSessionId = await resolveGameSessionId(roomCodeFromQuery)
                if (resolvedSessionId) {
                  setSessionId(resolvedSessionId)
                  console.log("[Leaderboard] Resolved game session ID:", resolvedSessionId)
                } else {
                  console.warn("[Leaderboard] No session found for game_pin:", roomCodeFromQuery)
                }
              } catch (err) {
                console.error("[Leaderboard] Supabase error:", err)
              }
            }
          }
        }

        if (!verified) {
          const urlParams = new URLSearchParams(window.location.search)
          const urlHostId = urlParams.get("hostId")
          const savedHostId = sessionStorage.getItem("currentHostId") || localStorage.getItem("currentHostId") || urlHostId

          if (savedHostId && roomCodeFromQuery) {
            console.log(`[Leaderboard] Auto-healing host session for ${roomCodeFromQuery} with hostId ${savedHostId}`)
            verified = true
          }
        }
      } catch (e) {
        console.error("Verification error", e)
      }

      if (verified) {
        setRoomCode(roomCodeFromQuery)
        setResolvingCode(false)
      } else {
        console.warn(`[Leaderboard] Access denied for room ${roomCodeFromQuery}`)
        router.push("/")
      }
    }

    verifyHostAccess()
  }, [roomCodeFromQuery, router])

  useEffect(() => {
    if (!roomCode && !resolvingCode) {
      router.push("/")
      return
    }

    const fetchRoom = async () => {
      try {
        const roomData = roomCode ? await roomManager.getRoom(roomCode) : null
        if (roomData) {
          setRoom(roomData as LeaderboardRoom)
          setHostId(roomData.hostId)
        } else if (!resolvingCode) {
          router.push("/")
        }
      } catch (error) {
        if (!resolvingCode) {
          router.push("/")
        }
      } finally {
        setLoading(false)
      }
    }

    fetchRoom()
  }, [roomCode, resolvingCode, router])

  useEffect(() => {
    if (!roomCode || loading) return
    let cancelled = false
    let unsub: (() => void) | undefined
    roomManager.subscribe(roomCode, (updatedRoom) => {
      if (updatedRoom) setRoom(updatedRoom as LeaderboardRoom)
    }).then((u) => {
      if (cancelled) u?.()
      else unsub = u
    })
    return () => {
      cancelled = true
      unsub?.()
    }
  }, [roomCode, loading])

  if (loading) {
    return <LeaderboardLoadingScreen title="MEMUAT PAPAN PERINGKAT..." subtitle="Menyiapkan hasil turnamen" />
  }

  if (!room) {
    return <LeaderboardSimpleScreen title="Room tidak ditemukan" subtitle="Mengarahkan ke beranda..." />
  }

  if (room.status !== "finished") {
    return <LeaderboardSimpleScreen showClock title="Game masih berlangsung" subtitle="Skor &amp; peringkat akan ditampilkan setelah permainan selesai." />
  }

  const playersToUse = room?.players.filter(p => !p.isHost) || []
  const getScoreStatusColor = (player: any) => {
    const score = Math.min(player.quizScore || 0, 100)
    return score < 75 ? 'low' : 'high'
  }
  const sortedPlayers = sortPlayersForLeaderboard(playersToUse, room)
    .map(player => ({
      ...player,
      quizScore: Math.min(player.quizScore || 0, 100)
    }))

  const champion = sortedPlayers[0]

  const resolvedHostId =
    hostId ||
    (typeof window !== 'undefined'
      ? (sessionStorage.getItem('currentHostId') || localStorage.getItem('currentHostId') || localStorage.getItem('hostId'))
      : null)
  const resolvedQuizId =
    room.quizId ||
    (typeof window !== 'undefined'
      ? (localStorage.getItem('quizId') || localStorage.getItem('selectedQuizId'))
      : null)
  const resolvedQuizTitle = room.quizTitle || 'Quiz'

  const handleRestart = async () => {
    if (!room || !resolvedHostId || !resolvedQuizId || isRestarting) return
    setIsRestarting(true)
    try {
      console.log('[Leaderboard] Restart dengan resolved hostId/quizId/quizTitle:',
        { hostId: resolvedHostId, quizId: resolvedQuizId, quizTitle: resolvedQuizTitle })
      const validGameMode = (room.settings.gameMode === 'easy' || room.settings.gameMode === 'medium' || room.settings.gameMode === 'hard')
        ? room.settings.gameMode as 'easy' | 'medium' | 'hard'
        : undefined

      const newRoom = await roomManager.createRoom(
        resolvedHostId,
        {
          questionCount: room.settings.questionCount,
          totalTimeLimit: room.settings.totalTimeLimit,
          gameMode: validGameMode,
        },
        resolvedQuizId,
        resolvedQuizTitle
      )
      if (!newRoom) {
        setIsRestarting(false)
        return
      }
      const verifyRoom = await roomManager.getRoom(newRoom.code)
      if (!verifyRoom) {
        setIsRestarting(false)
        return
      }
      try {
        await sessionManager.createOrUpdateSession(
          null,
          'host',
          {
            id: resolvedHostId,
            roomCode: newRoom.code,
            quizId: resolvedQuizId,
          },
          newRoom.code
        )
      } catch { }
      try {
        localStorage.setItem('hostId', resolvedHostId)
        localStorage.setItem('roomCode', newRoom.code)
        localStorage.setItem('quizId', resolvedQuizId)
        localStorage.setItem('currentHost', JSON.stringify({
          hostId: resolvedHostId,
          roomCode: newRoom.code,
          quizId: resolvedQuizId,
          isHost: true,
          timestamp: Date.now()
        }))
      } catch { }
      router.push(`/host/${newRoom.code}`)
    } catch (error) {
      setIsRestarting(false)
    }
  }

  const openStatistics = async () => {
    console.log("[Stats] Current roomCode:", roomCode)

    if (!roomCode) {
      alert('Kode room tidak tersedia')
      return
    }

    const gameSessionId = await resolveGameSessionId(roomCode)

    if (gameSessionId) {
      console.log("[Stats] ✅ Found game session ID:", gameSessionId)
      setSessionId(gameSessionId)
      window.open(`${STATS_BASE_URL}/${gameSessionId}`, '_blank')
    } else {
      console.error("[Stats] No session found for room:", roomCode)
      alert(`Sesi tidak ditemukan untuk room ${roomCode}`)
    }
  }

  const restartDisabled = isRestarting || !resolvedHostId || !resolvedQuizId

  return (
    <GameBackground>
      <FloatingActionButtons
        restartDisabled={restartDisabled}
        showHomeTooltip={showHomeTooltip}
        showStatTooltip={showStatTooltip}
        showRestartTooltip={showRestartTooltip}
        onHome={() => router.push("/")}
        onRestart={handleRestart}
        onStats={openStatistics}
        onHomeTooltip={setShowHomeTooltip}
        onStatTooltip={setShowStatTooltip}
        onRestartTooltip={setShowRestartTooltip}
      />

      <PageCornerLogos />

      <div className="relative z-10 container mx-auto px-4 py-4 md:py-8 mt-16 md:mt-24 lg:mt-28 flex-1 flex flex-col min-h-0">
        <MobileLeaderboard players={sortedPlayers as any} getScoreStatusColor={getScoreStatusColor} />
        <DesktopLeaderboard players={sortedPlayers as any} champion={champion} getScoreStatusColor={getScoreStatusColor} />
      </div>
    </GameBackground>
  )
}

export default function LeaderboardHostPage() {
  return (
    <>
      <PageTitle title={`Leaderboard`} />
      <Suspense fallback={<LeaderboardLoadingScreen title="MEMUAT PAPAN PERINGKAT..." subtitle="Menyiapkan hasil turnamen" />}>
        <LeaderboardHostPageContent />
      </Suspense>
    </>
  )
}
