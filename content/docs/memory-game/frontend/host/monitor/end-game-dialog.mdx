Berikut kode lengkap komponen ini dari proyek Memory Game — salin apa adanya ke jalur yang sama di proyek Anda.

#### app/host/[roomCode]/monitor/components/end-game-dialog.tsx

```tsx
import { X } from "lucide-react"
import { AlertDialog, AlertDialogTrigger, AlertDialogContent, AlertDialogHeader, AlertDialogTitle, AlertDialogDescription, AlertDialogFooter, AlertDialogCancel, AlertDialogAction } from "@/components/ui/alert-dialog"

interface EndGameDialogProps {
  isEndingGame: boolean
  onEnd: () => void
  compact?: boolean
  showLabel?: boolean
}

export function EndGameDialog({ isEndingGame, onEnd, compact = false, showLabel = true }: EndGameDialogProps) {
  return (
    <AlertDialog>
      <AlertDialogTrigger className={compact ? "shrink-0" : "h-full"}>
        <div
          className={compact
            ? "flex h-10 min-w-10 items-center justify-center gap-2 rounded-xl border border-red-500/50 bg-red-500/20 px-3 transition-all duration-200 hover:bg-red-500/30 cursor-pointer"
            : "h-full flex items-center gap-2.5 bg-red-500/20 border-2 border-red-500/50 rounded-lg px-5 hover:bg-red-500/30 transition-all duration-200 cursor-pointer"
          }
        >
          <div className={compact ? "w-4 h-4 flex items-center justify-center" : "w-5 h-5 flex items-center justify-center"}>
            <X className={compact ? "h-4 w-4 text-red-400 shrink-0" : "w-5 h-5 text-red-400"} strokeWidth={3} />
          </div>
          {showLabel && (
            <span className={compact ? "text-red-400 text-sm font-black pixel-font" : "text-red-400 font-bold text-xl pixel-font"}>
              AKHIRI GAME
            </span>
          )}
        </div>
      </AlertDialogTrigger>
      <AlertDialogContent className="bg-slate-900 border-2 border-white/20 text-white">
        <AlertDialogHeader>
          <AlertDialogTitle>Akhiri Game?</AlertDialogTitle>
          <AlertDialogDescription className="text-slate-300">Skor akan difinalisasi dan game akan berakhir.</AlertDialogDescription>
        </AlertDialogHeader>
        <AlertDialogFooter>
          <AlertDialogCancel className="bg-slate-800 text-white border-slate-600 hover:bg-slate-700 hover:text-white">Batal</AlertDialogCancel>
          <AlertDialogAction
            disabled={isEndingGame}
            onClick={(e) => {
              e.preventDefault()
              onEnd()
            }}
            className={`bg-red-600 hover:bg-red-700 text-white border-none disabled:opacity-50 disabled:cursor-not-allowed disabled:pointer-events-none ${isEndingGame ? 'opacity-50 cursor-not-allowed pointer-events-none' : ''}`}
          >
            {isEndingGame ? "..." : "Akhiri Game"}
          </AlertDialogAction>
        </AlertDialogFooter>
      </AlertDialogContent>
    </AlertDialog>
  )
}
```
