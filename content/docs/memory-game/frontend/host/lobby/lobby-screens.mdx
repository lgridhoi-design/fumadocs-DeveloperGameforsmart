interface LobbyCountdownProps {
  room: Room | null
  hostId: string | null
  onCountdownComplete: () => void
}

export function LobbyCountdown({ room, hostId, onCountdownComplete }: LobbyCountdownProps) {
  if (!room) return null
  return (
    <CountdownTimer
      room={room}
      playerId={hostId || undefined}
      isHost={true}
      onCountdownComplete={onCountdownComplete}
    />
  )
}
