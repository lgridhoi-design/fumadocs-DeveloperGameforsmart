          availableWidth * 0.9,
          availableHeight * 0.9,
          4000
        )
        setQrSize(Math.max(400, Math.floor(size)))
      }
      updateQrSize()
      const resizeTimer = setTimeout(updateQrSize, 100)
      window.addEventListener('resize', updateQrSize)
      return () => {
        window.removeEventListener('resize', updateQrSize)
        clearTimeout(resizeTimer)
      }
    }
  }, [showQRModal])

  const copyRoomCode = async () => {
    if (!roomCode) return;
    const textToCopy = roomCode;

    try {
      await navigator.clipboard.writeText(textToCopy);
      toast({
        title: t('lobby.roomCodeCopied') || t('lobby.shareCodeDesc'),
        description: t('lobby.shareCodeDesc'),
      });
      setCopiedCode(true);
      setTimeout(() => setCopiedCode(false), 1500);
    } catch (err) {
      console.error("Failed to copy with navigator.clipboard: ", err);
      try {
        const textArea = document.createElement("textarea");
        textArea.value = textToCopy;
        textArea.style.position = "fixed";
        textArea.style.left = "-9999px";
        document.body.appendChild(textArea);
        textArea.focus();
        textArea.select();
        document.execCommand("copy");
        document.body.removeChild(textArea);

        toast({
          title: t('lobby.roomCodeCopied') || t('lobby.shareCodeDesc'),
          description: t('lobby.shareCodeDesc'),
        });
        setCopiedCode(true);
        setTimeout(() => setCopiedCode(false), 1500);
      } catch (execErr) {
        console.error("Failed to copy with execCommand: ", execErr);
        toast({
          title: t('lobby.failedToCopy'),
          description: t('lobby.failedToCopyDesc'),
          variant: "destructive",
        });
      }
    }
  };

  const copyShareLink = async () => {
    if (!shareUrl) return;

    try {
      await navigator.clipboard.writeText(shareUrl);
      toast({
        title: t('lobby.shareLinkCopied'),
        description: t('lobby.shareLinkDesc'),
      });
      setCopiedLink(true);
      setTimeout(() => setCopiedLink(false), 1500);
    } catch (err) {
      console.error("Failed to copy with navigator.clipboard: ", err);
      try {
        const textArea = document.createElement("textarea");
        textArea.value = shareUrl;
        textArea.style.position = "fixed";
        textArea.style.left = "-9999px";
        document.body.appendChild(textArea);
        textArea.focus();
        textArea.select();
        document.execCommand("copy");
        document.body.removeChild(textArea);

        toast({
          title: t('lobby.shareLinkCopied'),
          description: t('lobby.shareLinkDesc'),
        });
        setCopiedLink(true);
        setTimeout(() => setCopiedLink(false), 1500);
      } catch (execErr) {
        console.error("Failed to copy with execCommand: ", execErr);
        toast({
          title: t('lobby.failedToCopy'),
          description: t('lobby.failedToCopyDesc'),
          variant: "destructive",
        });
      }
    }
  };

  const startGame = async () => {
    if (gameStarted || !currentRoom || !roomCode || !hostId) {
      return
    }

    try {
      const countdownSuccess = await roomManager.startCountdown(roomCode, hostId, 10)

      if (countdownSuccess) {
        const countdownStartTime = new Date().toISOString()
        setGameStarted(true)

        const updatedRoom = {
          ...currentRoom,
          status: "countdown" as const,
          countdownStartTime: countdownStartTime,
          countdownDuration: 10
        }

        setRoom(updatedRoom)

        if (typeof window !== 'undefined') {
          try {
            const countdownChannel = new BroadcastChannel(`countdown-${roomCode}`)
            const roomChannel = new BroadcastChannel(`room-${roomCode}`)

            const message = {
              type: 'countdown-started',
              room: updatedRoom,
              countdownStartTime: countdownStartTime,
              countdownDuration: 10
            }

            countdownChannel.postMessage(message)
            roomChannel.postMessage(message)

            setTimeout(() => {
              countdownChannel.close()
              roomChannel.close()
            }, 100)
          } catch (e) {
            console.error('[Lobby] Broadcast failed:', e)
          }
        }

        setTimeout(() => { }, 100)
      } else {
        toast({
          title: t('lobby.failedToStartGame'),
          description: t('lobby.couldNotStartCountdown'),
          duration: 3000,
        })
      }
    } catch (error) {
      toast({
        title: t('lobby.error'),
        description: t('lobby.errorStartingGame'),
        duration: 3000,
      })
    }
  }

  if (currentRoom && (currentRoom.status === "countdown" || (currentRoom.countdownStartTime && !currentRoom.gameStarted))) {
    return (
      <LobbyCountdown
        room={currentRoom}
        hostId={hostId}
        onCountdownComplete={handleCountdownComplete}
      />
    )
  }

  if (isRedirecting) {
    return <RedirectingScreen t={t} />
  }

  if (!roomCode || (loading && !currentRoom)) {
    return <LoadingScreen t={t} />
  }

  if (!currentRoom && !loading && !roomCode) {
    return <InitializingScreen t={t} />
  }

  if (!currentRoom && !loading) {
    return <RoomNotFoundScreen t={t} onBackToSelect={() => router.push("/select-quiz")} />
  }

  const handleExit = () => {
    if (quizId) localStorage.setItem("selectedQuizId", quizId)
    handleNavigationAttempt("/select-quiz")
  }

  return (
    <div className="min-h-screen relative overflow-hidden" style={{ background: 'linear-gradient(45deg, #1a1a2e, #16213e, #0f3460, #533483)' }}>

      <div className={`absolute inset-0 opacity-20 ${showQRModal ? 'blur-md' : ''}`}>
        <div className="pixel-grid"></div>
      </div>

      <div className={`absolute inset-0 opacity-10 ${showQRModal ? 'blur-md' : ''}`}>
        <div className="scanlines"></div>
      </div>

      <div className={`absolute inset-0 overflow-hidden ${showQRModal ? 'blur-md' : ''}`}>
        <PixelBackgroundElements />
      </div>

      <div className={showQRModal ? 'blur-md' : ''}>
        <PageCornerLogos memoryQuizNoLink />
      </div>

      <div className={`relative z-10 mx-auto max-w-7xl px-4 sm:px-6 py-4 sm:py-6 ${pageCornerLogoContentPadding} ${showQRModal ? 'blur-md' : ''}`}>
        <div className="mx-auto grid grid-cols-1 lg:grid-cols-2 gap-6 sm:gap-8 items-start">
          <RoomShareCard
            t={t}
            roomCode={roomCode}
            currentRoom={currentRoom}
            gameStarted={gameStarted}
            copiedCode={copiedCode}
            copiedLink={copiedLink}
            joinUrl={joinUrl}
            onCopyRoomCode={copyRoomCode}
            onCopyShareLink={copyShareLink}
            onShowQr={() => setShowQRModal(true)}
            onStartGame={startGame}
          />

          <PlayerListCard
            t={t}
            currentRoom={currentRoom}
            currentPlayers={currentPlayers}
            hostId={hostId}
            roomCode={roomCode}
            quizId={quizId}
            gameStarted={gameStarted}
            playerCountChanged={playerCountChanged}
            playerLeft={playerLeft}
            totalPages={totalPages}
            currentPage={currentPage}
            onInviteGroups={() => { fetchUserGroups(); setInviteGroupsOpen(true) }}
            onInviteFriends={() => { fetchFriends(); setInviteFriendsOpen(true) }}
            onExit={handleExit}
            onStartGame={startGame}
            onKickPlayer={handleKickPlayer}
            onNextPage={goToNextPage}
            onPrevPage={goToPreviousPage}
          />
        </div>
      </div>

      <LobbyDialogs
        t={t}
        inviteFriendsOpen={inviteFriendsOpen}
        setInviteFriendsOpen={setInviteFriendsOpen}
        loadingFriends={loadingFriends}
        friendsList={friendsList}
        friendSearchTerm={friendSearchTerm}
        setFriendSearchTerm={setFriendSearchTerm}
        invitedFriends={invitedFriends}
        onInviteFriend={inviteFriend}
        inviteGroupsOpen={inviteGroupsOpen}
        setInviteGroupsOpen={setInviteGroupsOpen}
        loadingGroups={loadingGroups}
        userGroups={userGroups}
        groupSearchTerm={groupSearchTerm}
        setGroupSearchTerm={setGroupSearchTerm}
        invitedGroups={invitedGroups}
        onInviteGroup={inviteGroup}
        showLeaveDialog={showLeaveDialog}
        setShowLeaveDialog={setShowLeaveDialog}
        isLeavingLobby={isLeavingLobby}
        onCancelLeave={cancelLeave}
        onConfirmLeave={confirmLeave}
        showKickDialog={showKickDialog}
        setShowKickDialog={setShowKickDialog}
        playerToKick={playerToKick}
        onConfirmKick={confirmKickPlayer}
        inviteToast={inviteToast}
        onHideToast={hideInviteToast}
        showQRModal={showQRModal}
        setShowQRModal={setShowQRModal}
        joinUrl={joinUrl}
      />

      {!showQRModal && (
        <FullscreenButton position="bottom-right" offsetX={16} offsetY={16} className="hidden lg:block" />
      )}
    </div>
  )
}
