Berikut kode lengkap komponen ini dari proyek Memory Game — salin apa adanya ke jalur yang sama di proyek Anda.

#### app/host/[roomCode]/lobby/components/room-share-card.tsx

```tsx
"use client"

import { Copy, Check, Maximize2, Play } from "lucide-react"
import { QRCodeSVG } from "qrcode.react"
import type { Room } from "@/lib/room-manager"

interface RoomShareCardProps {
  t: (key: string, opts?: any) => string
  roomCode: string | null
  currentRoom: Room | null
  gameStarted: boolean
  copiedCode: boolean
  copiedLink: boolean
  joinUrl: string
  onCopyRoomCode: () => void
  onCopyShareLink: () => void
  onShowQr: () => void
  onStartGame: () => void
}

export function RoomShareCard({
  t,
  roomCode,
  currentRoom,
  gameStarted,
  copiedCode,
  copiedLink,
  joinUrl,
  onCopyRoomCode,
  onCopyShareLink,
  onShowQr,
  onStartGame,
}: RoomShareCardProps) {
  return (
    <div className="relative pixel-lobby-container">
      <div className="absolute inset-0 bg-linear-to-br from-blue-600 to-purple-600 rounded-lg transform rotate-1 pixel-button-shadow"></div>
      <div className="relative bg-linear-to-br from-blue-500 to-purple-500 rounded-lg border-2 sm:border-4 border-black shadow-2xl pixel-lobby-card">
        <div className="p-4 sm:p-6 space-y-2">
          <div className="flex flex-row justify-center gap-2 sm:gap-4 mb-4">
            <div className="bg-blue-100 border border-blue-300 rounded-full px-3 sm:px-4 py-2 flex items-center gap-2 shadow-sm">
              <div className="w-4 h-4 sm:w-5 sm:h-5 bg-blue-500 rounded-full flex items-center justify-center">
                <span className="text-white text-xs">⏱️</span>
              </div>
              <span className="text-blue-700 font-bold text-[10px] pixel-font">
                {currentRoom?.settings.totalTimeLimit || 30}:00
              </span>
            </div>

            <div className="bg-green-100 border border-green-300 rounded-full px-3 sm:px-4 py-2 flex items-center gap-2 shadow-sm">
              <div className="w-4 h-4 sm:w-5 sm:h-5 bg-green-500 rounded-full flex items-center justify-center">
                <span className="text-white text-xs">❓</span>
              </div>
              <span className="text-green-700 font-bold text-[10px] pixel-font">
                {currentRoom?.settings.questionCount || 10} {t('lobby.questions')}
              </span>
            </div>
          </div>

          <div className="hidden md:block space-y-4">
            <div className="bg-white border-2 border-black rounded py-2 px-2 relative flex items-center justify-center min-h-[120px] overflow-hidden">
              <button
                onClick={onCopyRoomCode}
                className={`absolute right-2 top-1/2 -translate-y-1/2 w-12 h-12 rounded border-2 border-black flex items-center justify-center transition-all active:scale-95 z-30 ${copiedCode ? "bg-green-500 text-white" : "bg-gray-200 hover:bg-gray-300"}`}
              >
                {copiedCode ? <span className="font-bold sm:text-base md:text-xl">✓</span> : <Copy className="sm:h-5 sm:w-5 md:h-6 md:w-6" />}
              </button>
              <div className="text-center w-full pr-14 pl-2">
                <div className="pixel-font text-black font-bold !text-[50px] md:!text-[60px] lg:!text-[70px] xl:!text-[75px] leading-none whitespace-nowrap tracking-tighter">
                  {roomCode}
                </div>
              </div>
            </div>

            {joinUrl && (
              <div className="bg-white border-2 border-black rounded p-3 md:p-4">
                <div className="relative inline-block w-full max-w-full">
                  <div className="bg-white text-black rounded-lg py-3 px-4 w-full flex flex-col justify-center items-center relative">
                    <button
                      onClick={onShowQr}
                      className="absolute top-1 right-1 p-0.5 hover:bg-gray-100 rounded transition-colors z-10 border border-black w-6 h-6 flex items-center justify-center bg-white/80"
                    >
                      <Maximize2 className="w-3 h-3 text-gray-600" />
                    </button>

                    <div className="mb-3 py-0.5 w-full flex justify-center">
                      <QRCodeSVG
                        value={joinUrl}
                        size={typeof window !== 'undefined' ? Math.min(window.innerWidth * 0.3, 300) : 300}
                        className="mx-auto"
                        style={{ width: '100%', maxWidth: '300px', height: 'auto' }}
                      />
                    </div>

                    <div className="w-full">
                      <div className="flex items-center gap-2 bg-gray-100 rounded-lg p-3 border-2 border-black rtl:flex-row-reverse">
                        <span className="text-[10px] pixel-font break-all flex-1 text-gray-900 leading-relaxed font-bold">{joinUrl}</span>
                        <button
                          onClick={onCopyShareLink}
                          className="p-2 hover:bg-gray-200 rounded transition-colors shrink-0 border border-black flex items-center justify-center"
                        >
                          {copiedLink ? <Check className="w-4 h-4 text-green-600" /> : <Copy className="w-4 h-4 text-gray-600" />}
                        </button>
                      </div>
                    </div>
                  </div>
                </div>
              </div>
            )}
          </div>

          <div className="block md:hidden space-y-2">
            <div className="flex gap-2 items-stretch h-[70px]">
              <div className="bg-white border-2 border-black rounded relative flex items-center justify-center flex-1 overflow-hidden">
                <button
                  onClick={onCopyRoomCode}
                  className={`absolute right-1 top-1/2 -translate-y-1/2 w-8 h-8 rounded border-2 border-black flex items-center justify-center transition-all active:scale-95 z-30 ${copiedCode ? "bg-green-500 text-white" : "bg-gray-200"}`}
                >
                  {copiedCode ? <span className="font-bold text-sm">✓</span> : <Copy className="h-4 w-4" />}
                </button>
                <div className="text-center w-full pr-10 pl-2">
                  <div className="pixel-font text-black font-bold !text-[24px] leading-none whitespace-nowrap tracking-tighter">
                    {roomCode}
                  </div>
                </div>
              </div>
              {joinUrl && (
                <div
                  className="bg-white border-2 border-black rounded p-1 flex items-center justify-center shrink-0 w-[70px] relative"
                  onClick={onShowQr}
                  role="button"
                >
                  <div className="w-full h-full relative">
                    <QRCodeSVG
                      value={joinUrl}
                      size={60}
                      style={{ width: '100%', height: '100%' }}
                    />
                  </div>
                </div>
              )}
            </div>

            {joinUrl && (
              <div className="flex items-center gap-1 bg-white rounded-lg p-1.5 border-2 border-black rtl:flex-row-reverse shadow-sm">
                <span className="text-[8px] pixel-font break-all flex-1 text-gray-900 leading-tight font-bold pl-1">{joinUrl}</span>
                <button
                  onClick={onCopyShareLink}
                  className="p-1.5 hover:bg-gray-200 rounded transition-colors shrink-0 border border-black flex items-center justify-center bg-gray-100"
                >
                  {copiedLink ? <Check className="w-3 h-3 text-green-600" /> : <Copy className="w-3 h-3 text-gray-600" />}
                </button>
              </div>
            )}

            {currentRoom && !gameStarted && (
              <div className="w-full mt-2">
                <div className="relative pixel-button-container w-full">
                  <div className="absolute inset-0 bg-linear-to-br from-purple-600 to-purple-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
                  <button
                    onClick={onStartGame}
                    disabled={!roomCode || !currentRoom || currentRoom.players.length === 0}
                    className="relative bg-linear-to-br from-purple-500 to-purple-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-purple-400 hover:to-purple-500 transform hover:scale-105 transition-all duration-200 font-bold px-3 py-3 flex items-center justify-center gap-2.5 disabled:opacity-50 disabled:cursor-not-allowed w-full text-center"
                  >
                    <Play className="h-5 w-5 shrink-0" />
                    <span className="pixel-font text-sm whitespace-nowrap">{t('lobby.startGame')}</span>
                  </button>
                </div>
              </div>
            )}
          </div>
        </div>
      </div>
    </div>
  )
}
```
