#### app/player/[roomCode]/game/memory-challenge/page.tsx

```tsx
"use client"

import { useState, useEffect } from "react"
import { useRouter } from "next/navigation"
import { MemoryGame } from "@/components/game/memory-game"
import { roomManager } from "@/lib/room-manager"
import { supabaseRoomManager } from "@/lib/supabase-room-manager"
import { useRoom } from "@/hooks/use-room"
import { sessionManager } from "@/lib/supabase-session-manager"
import {
  canAccessMemoryChallenge,
  getQuizProgress,
  saveQuizProgress,
  type QuizMemoryProgress
} from "@/lib/quiz-memory-flow"
import { supabase } from "@/lib/supabase"
import { isPlayersSupabaseConfigured, supabasePlayers } from "@/lib/supabase-players"
import { getServerNow, syncServerTime } from "@/lib/server-time"
import { MemoryGameShell, MemoryStatusScreen } from "./components/memory-game-shell"
import { MemoryCompletedScreen } from "./components/memory-completed-screen"

interface MemoryChallengePageProps {
  params: { roomCode: string }
}

export default function MemoryChallengePage({ params }: MemoryChallengePageProps) {
  const router = useRouter()

  const [correctMatches, setCorrectMatches] = useState(0)
  const [gameCompleted, setGameCompleted] = useState(false)
  const [timeElapsedMs, setTimeElapsedMs] = useState(Number.MAX_SAFE_INTEGER)
  const [hasAccess, setHasAccess] = useState(false)
  const [loading, setLoading] = useState(true)

  const [playerId, setPlayerId] = useState<string | null>(null)
  const [playerData, setPlayerData] = useState<any>(null)
  const { room } = useRoom(params.roomCode)
  const gameMode = room?.settings?.gameMode || "easy"
  const targetMatches = gameMode === "hard" ? 10 : gameMode === "medium" ? 8 : 6

  useEffect(() => {
    const loadPlayerData = async () => {
      try {
        const sessionId = sessionManager.getSessionIdFromStorage()
        if (sessionId) {
          const sessionData = await sessionManager.getSessionData(sessionId)
          if (sessionData && sessionData.user_type === 'player') {
            setPlayerId(sessionData.user_data.id)
            setPlayerData(sessionData.user_data)
          }
        }
        if (typeof window !== 'undefined') {
          const player = localStorage.getItem("currentPlayer")
          if (player) {
            const playerInfo = JSON.parse(player)
            setPlayerId(playerInfo.id)
            setPlayerData(playerInfo)
          }
        }
      } catch (error) {
        console.error("[Memory Challenge] Error loading player data:", error)
      }
    }
    loadPlayerData()
  }, [])

  useEffect(() => {
    if (!playerId || !params.roomCode) return

    const client = isPlayersSupabaseConfigured() ? supabasePlayers : supabase
    const broadcastChannel = client.channel(`progress-${params.roomCode}`)

    const updateDbPhase = async () => {
      try {
        await supabaseRoomManager.updateGameProgress(params.roomCode, playerId, { phase: "GAME" })
      } catch (dbError) {
        console.warn('[Memory Challenge] Failed to update phase in DB:', dbError)
      }
    }
    updateDbPhase()

    broadcastChannel.subscribe((status) => {
      if (status === 'SUBSCRIBED') {
        const nickname = playerData?.nickname ||
          (typeof window !== 'undefined' && localStorage.getItem("currentPlayer")
            ? JSON.parse(localStorage.getItem("currentPlayer") || "{}").nickname : "")
        broadcastChannel.send({
          type: 'broadcast',
          event: 'progress-update',
          payload: { playerId, updateData: { nickname, phase: "GAME" }, timestamp: Date.now() }
        }).catch(() => {})
      }
    })

    return () => {
      client.removeChannel(broadcastChannel)
    }
  }, [playerId, params.roomCode, playerData])

  useEffect(() => {
    if (room && room.status === "finished") {
      router.push(`/player/${params.roomCode}/result`)
    }
  }, [room?.status, params.roomCode])

  useEffect(() => {
    const loadProgress = async () => {
      if (!playerId) return
      try {
        const savedCardsState = localStorage.getItem(`memory-cards-state-${params.roomCode}`)
        if (savedCardsState) {
          const parsed = JSON.parse(savedCardsState)
          if (parsed.cards && Array.isArray(parsed.cards)) {
            const matchedCount = parsed.cards.filter((c: any) => c.isMatched).length / 2
            setCorrectMatches(matchedCount)
            if (matchedCount >= (parsed.cards.length / 2)) setGameCompleted(true)
            return
          }
        }
        setCorrectMatches(0)
      } catch (error) {
        console.error("[Memory Challenge] Error loading progress:", error)
        setCorrectMatches(0)
      }
    }
    loadProgress()
  }, [params.roomCode, playerId])

  useEffect(() => {
    const checkAccess = () => {
      try {
        if (canAccessMemoryChallenge(params.roomCode)) {
          const progress = getQuizProgress(params.roomCode)
          if (progress) saveQuizProgress(params.roomCode, { ...progress, pendingMemoryGame: true })
          setHasAccess(true)
          setLoading(false)
          return
        }
        router.push(`/player/${params.roomCode}/quiz`)
      } catch (error) {
        console.error("[Memory Challenge] Error checking access:", error)
        router.push(`/player/${params.roomCode}/quiz`)
      }
    }
    if (playerId !== null) checkAccess()
  }, [params.roomCode, playerId])

  useEffect(() => {
    if (gameCompleted || !room) return
    syncServerTime()

    const interval = setInterval(() => {
      const now = getServerNow()
      if (!room.startedAt) {
        setTimeElapsedMs(Number.MAX_SAFE_INTEGER)
        return
      }
      const startedAtMs = new Date(room.startedAt).getTime()
      const totalTimeLimitSeconds = (room.settings?.totalTimeLimit ?? 0) * 60
      const elapsedMs = now - startedAtMs
      const remainingMs = Math.ceil(Math.max(0, (totalTimeLimitSeconds * 1000) - elapsedMs) / 1000) * 1000
      setTimeElapsedMs(remainingMs)
    }, 250)

    return () => clearInterval(interval)
  }, [gameCompleted, room])

  const handleGameEnd = async () => {
    const existing: QuizMemoryProgress = getQuizProgress(params.roomCode) || {
      currentQuestion: 0,
      correctAnswers: 0,
      quizScore: 0,
      questionsAnswered: 0,
      memoryGamesCompleted: 0,
      questionsSinceLastMemory: 0
    }

    const updatedProgress: QuizMemoryProgress = {
      ...existing,
      memoryGamesCompleted: existing.memoryGamesCompleted + 1,
      questionsSinceLastMemory: 0,
      pendingMemoryGame: false
    }

    saveQuizProgress(params.roomCode, updatedProgress)

    localStorage.setItem(`memory-return-${params.roomCode}`, JSON.stringify({
      score: 0,
      quizScore: 0,
      correctAnswers: 0,
      questionsAnswered: updatedProgress.questionsAnswered || 0,
      resumeQuestion: updatedProgress.currentQuestion || 0,
      memoryGamesCompleted: updatedProgress.memoryGamesCompleted,
      questionsSinceLastMemory: 0
    }))
    localStorage.removeItem(`memory-progress-${params.roomCode}`)
    localStorage.removeItem(`memory-cards-state-${params.roomCode}`)

    const totalQuestions = room?.settings.questionCount || 10
    const questionsAnswered = updatedProgress.questionsAnswered || 0

    if (questionsAnswered >= totalQuestions) {
      try {
        const playerData = localStorage.getItem("currentPlayer")
        const isHost = playerData ? JSON.parse(playerData).isHost : false

        if (isHost) {
          await roomManager.updateGameStatus(params.roomCode, "finished")
          router.push(`/host/leaderboard?roomCode=${params.roomCode}`)
        } else {
          if (playerId) {
            try {
              const gameStartMs = room?.startedAt ? new Date(room.startedAt).getTime() : 0
              const actualDurationSeconds = gameStartMs > 0
                ? Math.max(1, Math.floor((getServerNow() - gameStartMs) / 1000))
                : undefined

              await roomManager.markPlayerFinished(
                params.roomCode, playerId, undefined, questionsAnswered, undefined, actualDurationSeconds
              )

              try {
                await roomManager.maybeFinishGameIfLastPlayer(params.roomCode, playerId)
              } catch (lastErr) {
                console.warn('[Memory Challenge] maybeFinishGameIfLastPlayer error:', lastErr)
              }

              const raw = typeof window !== "undefined" ? localStorage.getItem("currentPlayer") : null
              const cp = raw ? JSON.parse(raw) : null
              if (typeof window !== "undefined") {
                localStorage.setItem(`quiz-last-result-${params.roomCode}`, JSON.stringify({
                  at: getServerNow(),
                  id: playerId,
                  nickname: cp?.nickname || "",
                  avatar: cp?.avatar || "",
                  quizScore: 0,
                  questionsAnswered: questionsAnswered,
                  correctAnswers: 0,
                  timeSpent: actualDurationSeconds || 0,
                  joinedAt: cp?.joinedAt || null,
                }))
              }
            } catch (err) {
              console.warn('[Memory Challenge] Error marking player finished:', err)
            }
          }
          router.push(`/player/${params.roomCode}/result`)
        }
      } catch (error) {
        console.error("[Memory Challenge] Error ending game:", error)
        router.push(`/player/${params.roomCode}/quiz`)
      }
    } else {
      router.push(`/player/${params.roomCode}/quiz`)
    }
  }

  useEffect(() => {
    if (gameCompleted) {
      const timeout = setTimeout(() => handleGameEnd(), 2000)
      return () => clearTimeout(timeout)
    }
  }, [gameCompleted])

  useEffect(() => {
    if (correctMatches > 0 && correctMatches >= targetMatches && !gameCompleted) {
      setGameCompleted(true)
    }
  }, [correctMatches, gameCompleted, targetMatches])

  const handleCorrectMatch = () => {
    const newCount = correctMatches + 1
    localStorage.setItem(`memory-progress-${params.roomCode}`, newCount.toString())
    setCorrectMatches(newCount)

    if (newCount >= targetMatches) setGameCompleted(true)
  }

  if (loading || !room) {
    return <MemoryStatusScreen iconClass="bg-linear-to-r from-blue-400 to-purple-400" title="Memvalidasi Akses..." />
  }
  if (!hasAccess) {
    return <MemoryStatusScreen iconClass="bg-linear-to-r from-red-400 to-orange-400" title="Akses Ditolak" />
  }

  return (
    <MemoryGameShell>
      {!gameCompleted ? (
        <div className="max-w-2xl mx-auto">
          <MemoryGame
            onCorrectMatch={handleCorrectMatch}
            disabled={gameCompleted}
            roomCode={params.roomCode}
            gameMode={room?.settings?.gameMode || "easy"}
            timeElapsedMs={timeElapsedMs}
          />
        </div>
      ) : (
        <MemoryCompletedScreen />
      )}
    </MemoryGameShell>
  )
}
```
