#### app/player/[roomCode]/quiz/components/quiz-screens.tsx

```tsx
"use client"

import { Users, Trophy } from "lucide-react"

const bg = { background: 'linear-gradient(45deg, #1a1a2e, #16213e, #0f3460, #533483)' }

interface QuizLoadingScreenProps {
  title: string
  subtitle?: string
  icon?: "users" | "trophy"
}

export function QuizLoadingScreen({ title, subtitle, icon = "users" }: QuizLoadingScreenProps) {
  return (
    <div className="min-h-screen relative overflow-hidden" style={bg}>
      <div className="absolute inset-0 opacity-20">
        <div className="pixel-grid"></div>
      </div>
      <div className="relative z-10 flex items-center justify-center min-h-screen">
        <div className="text-center">
          <div className="w-12 h-12 bg-linear-to-r from-blue-400 to-purple-400 rounded-lg border-2 border-white shadow-xl flex items-center justify-center pixel-brain mb-4 mx-auto animate-pulse">
            {icon === "trophy" ? (
              <Trophy className="w-6 h-6 text-white" />
            ) : (
              <Users className="w-6 h-6 text-white" />
            )}
          </div>
          <h2 className="text-xl font-bold text-white mb-2">{title}</h2>
          {subtitle && <p className="text-sm text-blue-200">{subtitle}</p>}
        </div>
      </div>
    </div>
  )
}
```
