#### app/player/[roomCode]/quiz/components/quiz-game-view.tsx

```tsx
"use client"

import { createPortal } from "react-dom"
import { Clock, Maximize2, X } from "lucide-react"
import type { Question } from "@/lib/quiz-data"
import { PageCornerLogos, pageCornerLogoContentPadding } from "@/components/layout/page-corner-logos"
import { PixelBackgroundElements } from "@/components/layout/pixel-background-elements"
import {
  isImageUrl,
  needsScrollableContainer,
  getQuestionFontSize,
  getQuestionAlignment,
  renderTextWithLineBreaks,
  getOptionFontSize,
} from "./quiz-utils"

type ShuffledOption = { shuffled: (string | { text: string; image?: string })[], originalIndices: number[] }

interface QuizGameViewProps {
  progress: number
  showTimeWarning: boolean
  isTimeUp: boolean
  timerDisplayText: string
  question: Question
  currentShuffled: ShuffledOption
  currentQuestion: number
  totalQuestions: number
  selectedAnswer: number | null
  zoomedImage: string | null
  onAnswerSelect: (index: number) => void
  onZoomedImageChange: (value: string | null) => void
}

export function QuizGameView({
  progress,
  showTimeWarning,
  isTimeUp,
  timerDisplayText,
  question,
  currentShuffled,
  currentQuestion,
  totalQuestions,
  selectedAnswer,
  zoomedImage,
  onAnswerSelect,
  onZoomedImageChange,
}: QuizGameViewProps) {
  return (
    <div className="min-h-screen relative overflow-hidden" style={{ background: 'linear-gradient(45deg, #1a1a2e, #16213e, #0f3460, #533483)' }}>
      <div className="absolute inset-0 opacity-20">
        <div className="pixel-grid"></div>
      </div>

      <div className="absolute inset-0 opacity-10">
        <div className="scanlines"></div>
      </div>

      <div className="absolute inset-0 overflow-hidden">
        <PixelBackgroundElements />
        <div className="absolute top-20 left-10 w-32 h-32 opacity-20 animate-float">
          <div className="w-full h-full rounded-full bg-linear-to-r from-blue-400 to-purple-400 blur-xl"></div>
        </div>
        <div className="absolute top-40 right-20 w-24 h-24 opacity-30 animate-float-delayed">
          <div className="w-full h-full rounded-full bg-linear-to-r from-cyan-400 to-blue-400 blur-lg"></div>
        </div>
        <div className="absolute bottom-32 left-1/4 w-40 h-40 opacity-25 animate-float-slow">
          <div className="w-full h-full rounded-full bg-linear-to-r from-purple-400 to-pink-400 blur-2xl"></div>
        </div>
        <div className="absolute bottom-20 right-1/3 w-28 h-28 opacity-35 animate-float-delayed-slow">
          <div className="w-full h-full rounded-full bg-linear-to-r from-green-400 to-cyan-400 blur-xl"></div>
        </div>
      </div>

      <PageCornerLogos />

      <div className={`relative z-10 container mx-auto px-4 py-8 ${pageCornerLogoContentPadding}`}>
        <div className="mb-6">
          <div className="w-full max-w-xl mx-auto">
            <div className="flex items-center justify-between mb-1">
              <span className="text-xs sm:text-sm font-bold text-white">PROGRES</span>
              <span className="text-xs sm:text-sm text-blue-300">{Math.round(progress)}%</span>
            </div>
            <div className="w-full bg-black/30 border border-white/30 rounded-lg h-2">
              <div
                className="h-full bg-linear-to-r from-blue-400 to-purple-400 rounded-lg transition-all duration-300"
                style={{ width: `${progress}%` }}
              />
            </div>
          </div>
        </div>

        {showTimeWarning && (
          <div className="mx-auto mb-6 w-full max-w-3xl rounded-2xl border border-red-500/50 bg-red-500/20 p-4 shadow-lg shadow-red-500/10 animate-pulse sm:p-5">
            <div className="flex items-center justify-center gap-2 text-center">
              <Clock className="h-4 w-4 text-red-400 sm:h-5 sm:w-5" />
              <span className="text-lg font-black uppercase tracking-[0.18em] text-red-400 sm:text-xl">
                {isTimeUp ? 'WAKTU HABIS!' : 'WAKTU HAMPIR HABIS!'}
              </span>
              <Clock className="h-4 w-4 text-red-400 sm:h-5 sm:w-5" />
            </div>
            <p className="mt-1 text-center text-xs text-red-200 sm:text-sm">
              {isTimeUp
                ? 'Game akan segera berakhir'
                : 'Game akan berakhir saat waktu habis'}
            </p>
          </div>
        )}

        <div className="flex justify-center gap-2 sm:gap-4 mb-4 sm:mb-6">
          <div className="bg-blue-500/20 border-2 border-blue-500/50 rounded-lg px-2 sm:px-4 py-1 sm:py-2 flex items-center gap-1 sm:gap-2">
            <span className="text-blue-400 font-bold text-[10px] sm:text-sm">Soal {currentQuestion + 1}/{totalQuestions}</span>
          </div>

          <div className={`${showTimeWarning ? 'bg-red-500/20 border-red-500/50 animate-pulse' : 'bg-green-500/20 border-green-500/50'} border-2 rounded-lg px-2 sm:px-4 py-1 sm:py-2 flex items-center gap-1 sm:gap-2`}>
            <Clock className={`w-3 h-3 sm:w-4 sm:h-4 ${showTimeWarning ? 'text-red-400' : 'text-green-400'}`} />
            <span className={`font-bold text-[10px] sm:text-sm ${showTimeWarning ? 'text-red-400' : 'text-green-400'}`}>
              {timerDisplayText}
            </span>
          </div>
        </div>

        <div className="max-w-3xl mx-auto">
          <div className="bg-linear-to-br from-white/10 to-white/5 border-2 border-white/30 rounded-lg p-6 pixel-lobby-card">
            <div className="text-center mb-6">
              <div className="flex flex-col">
                {(question.image || isImageUrl(question.question)) && (
                  <div className="flex justify-center mb-6 order-first relative group">
                    <img
                      src={question.image || question.question}
                      alt="Question"
                      className="max-h-64 max-w-full rounded-lg object-contain shadow-lg bg-black/20 cursor-zoom-in hover:brightness-110 transition-all"
                      onClick={() => onZoomedImageChange(question.image || question.question)}
                    />
                    <div className="absolute top-2 right-2 opacity-0 group-hover:opacity-100 transition-opacity bg-black/50 rounded-full p-1 pointer-events-none">
                      <Maximize2 className="w-4 h-4 text-white" />
                    </div>
                  </div>
                )}

                {!isImageUrl(question.question) && (
                  <div className={needsScrollableContainer(question.question) ? 'max-h-48 overflow-y-auto custom-scrollbar px-2 w-full' : 'px-2 w-full'}>
                    <h2 className={`font-bold text-white mb-3 leading-relaxed break-words ${getQuestionFontSize(question.question)} ${getQuestionAlignment(question.question, false)}`}>
                      {renderTextWithLineBreaks(question.question)}
                    </h2>
                  </div>
                )}
              </div>
            </div>
            <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
              {currentShuffled.shuffled.map((option, index) => {
                const isSelectedAnswer = index === selectedAnswer

                return (
                  <button
                    key={index}
                    className={`w-full p-4 text-start rounded-lg border-2 transition-all duration-300 hover:scale-105 ${
                      isSelectedAnswer
                        ? "bg-blue-500/20 border-blue-400 text-blue-300"
                        : "bg-white/5 border-white/20 text-white hover:bg-white/10"
                      }`}
                    onClick={() => onAnswerSelect(index)}
                    disabled={selectedAnswer !== null}
                  >
                    <div className="flex items-center gap-3">
                      <div className={`w-8 h-8 rounded border border-white flex items-center justify-center font-bold text-sm ${isSelectedAnswer
                        ? "bg-blue-400 text-white"
                        : "bg-white/20 text-white"
                        }`}>
                        {String.fromCharCode(65 + index)}
                      </div>
                      <div className="flex-1 flex flex-col justify-start gap-2 min-w-0">
                        {(() => {
                          const imgUrl = typeof option === 'object' ? option.image : (isImageUrl(option) ? option : null)

                          if (imgUrl) return (
                            <div className="relative group w-fit flex justify-start">
                              <img
                                src={imgUrl}
                                alt={`Option ${index + 1}`}
                                className="h-32 sm:h-40 w-auto rounded object-contain bg-white/5 cursor-zoom-in hover:brightness-110 transition-all"
                                onClick={(e) => {
                                  e.stopPropagation()
                                  onZoomedImageChange(imgUrl)
                                }}
                              />
                              <div className="absolute top-1 end-2 opacity-0 group-hover:opacity-100 transition-opacity bg-black/50 rounded-full p-1 pointer-events-none">
                                <Maximize2 className="w-3 h-3 text-white" />
                              </div>
                            </div>
                          )
                          return null
                        })()}

                        {(() => {
                          const text = typeof option === 'object' ? option.text : option
                          const isImg = typeof option === 'string' && isImageUrl(option)

                          if (text && (!isImg || (typeof option === 'object' && (option as any).text))) {
                            return <span className={`font-medium break-words text-start ${getOptionFontSize(text)}`}>{text}</span>
                          }
                          return null
                        })()}
                      </div>
                    </div>
                  </button>
                )
              })}
            </div>
          </div>
        </div>
      </div>

      <PixelBackgroundElements />

      {zoomedImage && typeof document !== 'undefined' && createPortal(
        <div
          className="fixed inset-0 z-[9999] flex items-center justify-center bg-black/90 backdrop-blur-sm p-4 animate-in fade-in duration-200"
          onClick={() => onZoomedImageChange(null)}
          style={{ position: 'fixed', top: 0, left: 0, right: 0, bottom: 0 }}
        >
          <div
            className="relative max-w-5xl max-h-[90vh] flex flex-col items-center justify-center"
            onClick={(e) => e.stopPropagation()}
          >
            <button
              className="absolute -top-12 right-0 md:-right-12 text-white bg-black/50 hover:bg-white/20 rounded-full p-2 transition-colors z-50"
              onClick={() => onZoomedImageChange(null)}
            >
              <X className="w-6 h-6" />
            </button>
            <img
              src={zoomedImage}
              alt="Zoomed"
              className="max-w-full max-h-[85vh] object-contain rounded-lg shadow-2xl"
            />
          </div>
        </div>,
        document.body
      )}
    </div>
  )
}
```
