"use client"

import { useState, useEffect } from "react"
import { useRouter } from "next/navigation"
import { Card, CardContent } from "@/components/ui/card"
import { useRoom } from "@/hooks/use-room"
import { CountdownTimer } from "@/components/game/countdown-timer"
import { sessionManager } from "@/lib/supabase-session-manager"
import { initQuizProgressForSession, memoryChallengePath } from "@/lib/quiz-memory-flow"
import { roomManager } from "@/lib/room-manager"
import { useFullscreenManager } from "@/hooks/use-fullscreen"
import { FullscreenPrompt } from "@/components/guards/fullscreen-prompt"

interface CountdownPageProps {
  params: { roomCode: string }
  searchParams: { quizId?: string; questionCount?: string; timeLimit?: string; mode?: string }
}

export default function CountdownPage({ params, searchParams }: CountdownPageProps) {
  const router = useRouter()
  const { room, loading } = useRoom(params.roomCode)
  const { isMobile, requestFullscreen } = useFullscreenManager()

  const [playerId, setPlayerId] = useState<string | null>(null)
  const [hostId, setHostId] = useState<string | null>(null)
  const [isHost, setIsHost] = useState(false)
  const [showFullscreenPrompt, setShowFullscreenPrompt] = useState(false)

  useEffect(() => {
    const loadPlayerData = async () => {
      try {
        const sessionId = sessionManager.getSessionIdFromStorage()
        if (sessionId) {
          const sessionData = await sessionManager.getSessionData(sessionId)
          if (sessionData) {
            setPlayerId(sessionData.user_data.id)
            if (sessionData.user_type === 'host') setHostId(sessionData.user_data.hostId)
          }
        }
        if (typeof window !== 'undefined') {
          const player = localStorage.getItem("currentPlayer")
          if (player) setPlayerId(JSON.parse(player).id)
        }
      } catch (error) {
        console.error("[Countdown] Error loading player data:", error)
      }
    }
    loadPlayerData()
  }, [])

  useEffect(() => {
    if (room && playerId) {
      if (room.hostId && hostId && room.hostId === hostId) {
        setIsHost(true)
      } else {
        setIsHost(room.players.some((p) => p.id === playerId && p.isHost))
      }
    }
  }, [room, playerId, hostId])

  useEffect(() => {
    if (!loading && (!room || !room.gameStarted)) router.push("/")
  }, [room, loading])

  const redirectToGame = () => {
    const isTryOut = searchParams.mode === 'tryout'
    if (isHost && !isTryOut) {
      const savedHostId = sessionStorage.getItem("currentHostId") || localStorage.getItem("currentHostId") || ""
      void roomManager.startGame(params.roomCode, savedHostId).catch(() => {
        void roomManager.updateGameStatus(params.roomCode, 'quiz')
      })
      router.replace(`/host/${params.roomCode}/monitor`)
    } else {
      if (isHost && isTryOut) {
        const savedHostId = sessionStorage.getItem("currentHostId") || localStorage.getItem("currentHostId") || ""
        void roomManager.startGame(params.roomCode, savedHostId).catch(() => {
          void roomManager.updateGameStatus(params.roomCode, 'quiz')
        })
      }
      initQuizProgressForSession(params.roomCode)
      router.replace(memoryChallengePath(params.roomCode))
    }
  }

  const handleCountdownComplete = () => {
    if (!isHost && isMobile && !document.fullscreenElement) {
      setShowFullscreenPrompt(true)
      return
    }
    redirectToGame()
  }

  const handleFullscreenTap = async () => {
    await requestFullscreen()
    redirectToGame()
  }

  if (loading) {
    return (
      <div className="min-h-screen bg-gradient-to-br from-background via-card to-muted flex items-center justify-center">
        <Card className="max-w-md mx-auto text-center">
          <CardContent className="py-8"><div className="text-lg">Memuat...</div></CardContent>
        </Card>
      </div>
    )
  }

  if (!room || !room.gameStarted) {
    return (
      <div className="min-h-screen bg-gradient-to-br from-background via-card to-muted flex items-center justify-center">
        <Card className="max-w-md mx-auto text-center">
          <CardContent className="py-8"><div className="text-lg">Sesi game tidak valid</div></CardContent>
        </Card>
      </div>
    )
  }

  if (showFullscreenPrompt) return <FullscreenPrompt onFullscreen={handleFullscreenTap} />

  if (room.status !== "countdown") {
    redirectToGame()
    return null
  }

  return <CountdownTimer room={room} playerId={playerId || undefined} isHost={isHost} onCountdownComplete={handleCountdownComplete} />
}
