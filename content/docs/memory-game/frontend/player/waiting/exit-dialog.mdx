Berikut kode lengkap komponen ini dari proyek Memory Game — salin apa adanya ke jalur yang sama di proyek Anda.

#### app/player/[roomCode]/waiting/components/exit-dialog.tsx

```tsx
import { ArrowLeft } from "lucide-react"
import { Button } from "@/components/ui/button"
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle, AlertDialogTrigger } from "@/components/ui/alert-dialog"

interface ExitDialogProps {
  open: boolean
  roomCode: string
  isLeaving: boolean
  onOpenChange: (open: boolean) => void
  onLeave: () => void
}

export function ExitDialog({ open, roomCode, isLeaving, onOpenChange, onLeave }: ExitDialogProps) {
  return (
    <div className="relative pixel-button-container">
      <div className="absolute inset-0 bg-linear-to-br from-red-600 to-red-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
      <AlertDialog open={open} onOpenChange={onOpenChange}>
        <AlertDialogTrigger asChild>
          <Button
            onClick={() => onOpenChange(true)}
            className="relative flex items-center bg-linear-to-br from-red-500 to-red-600 border-2 border-black rounded-lg px-2 py-1 text-[9px] font-bold text-white transition-all duration-200 hover:scale-105 sm:px-3 sm:py-1.5 md:px-3 md:py-2"
          >
            <ArrowLeft className="me-1 h-3 w-3 shrink-0" />
            <span className="hidden md:inline ml-1 pixel-font-sm">KELUAR</span>
          </Button>
        </AlertDialogTrigger>
        <AlertDialogContent className="bg-linear-to-br from-blue-500 to-purple-500 border-2 sm:border-4 border-black shadow-2xl pixel-dialog max-w-[90vw] sm:max-w-md">
          <AlertDialogHeader>
            <div className="text-center mb-3 sm:mb-4">
              <div className="inline-block bg-white border-2 border-black rounded px-3 py-1.5 sm:px-4 sm:py-2 mb-2 sm:mb-3">
                <AlertDialogTitle className="text-black font-bold text-sm sm:text-base md:text-lg pixel-font">KELUAR DARI RUANGAN?</AlertDialogTitle>
              </div>
            </div>
            <div className="bg-black/20 border border-white/30 rounded px-3 py-2 sm:px-4 sm:py-3 text-center">
              <AlertDialogDescription className="text-white text-xs sm:text-sm pixel-font-sm">
                Apakah Anda yakin ingin keluar dari ruangan ini?
              </AlertDialogDescription>
            </div>
          </AlertDialogHeader>
          <AlertDialogFooter className="flex flex-col sm:flex-row gap-2 sm:gap-3 justify-center mt-4 sm:mt-6">
            <div className="relative pixel-button-container w-full sm:w-auto">
              <div className="absolute inset-0 bg-linear-to-br from-gray-600 to-gray-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
              <AlertDialogCancel
                onClick={() => {
                  window.history.pushState({ waitingRoom: true, roomCode }, '', window.location.href)
                }}
                className="relative bg-linear-to-br from-gray-500 to-gray-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-gray-400 hover:to-gray-500 transform hover:scale-105 transition-all duration-200 font-bold pixel-font-sm w-full sm:w-auto px-4 py-2 sm:px-6 sm:py-3 min-h-[44px]"
              >
                BATAL
              </AlertDialogCancel>
            </div>
            <div className="relative pixel-button-container w-full sm:w-auto">
              <div className="absolute inset-0 bg-linear-to-br from-red-600 to-red-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
              <AlertDialogAction
                disabled={isLeaving}
                onClick={(e) => {
                  e.preventDefault()
                  onLeave()
                }}
                className={`relative bg-linear-to-br from-red-500 to-red-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-red-400 hover:to-red-500 transform hover:scale-105 transition-all duration-200 font-bold pixel-font-sm w-full sm:w-auto px-4 py-2 sm:px-6 sm:py-3 min-h-[44px] disabled:opacity-50 disabled:cursor-not-allowed disabled:pointer-events-none disabled:hover:scale-100 ${isLeaving ? 'opacity-50 cursor-not-allowed pointer-events-none' : ''}`}
              >
                {isLeaving ? "..." : "KELUAR"}
              </AlertDialogAction>
            </div>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  )
}
```
