```tsx
import { Crown, Clock } from "lucide-react"
import { GameBackground } from "@/components/layout/game-background"

interface LoadingScreenProps {
  title: string
  subtitle?: string
}

export function LeaderboardLoadingScreen({ title, subtitle }: LoadingScreenProps) {
  return (
    <GameBackground>
      <div className="relative z-10 flex items-center justify-center min-h-screen px-4">
        <div className="text-center w-full max-w-lg mx-auto">
          <div className="relative mb-6 sm:mb-8">
            <div className="absolute inset-0 bg-gradient-to-r from-yellow-400/40 to-amber-400/40 blur-2xl animate-pulse"></div>
            <div className="relative w-16 h-16 sm:w-20 sm:h-20 bg-gradient-to-r from-yellow-400 to-amber-500 rounded-full border-4 border-white shadow-2xl flex items-center justify-center mx-auto animate-bounce-slow">
              <Crown className="w-8 h-8 sm:w-10 sm:h-10 text-white" />
            </div>
          </div>
          <h2 className="text-xl sm:text-2xl md:text-3xl font-bold text-transparent bg-clip-text bg-gradient-to-r from-yellow-300 to-amber-400 mb-3 sm:mb-4 px-2 break-words leading-tight">{title}</h2>
          {subtitle && <p className="text-sm sm:text-base md:text-lg text-yellow-200 font-semibold px-2 break-words leading-normal">{subtitle}</p>}
        </div>
      </div>
    </GameBackground>
  )
}

interface SimpleScreenProps {
  title: string
  subtitle: string
  showClock?: boolean
}

export function LeaderboardSimpleScreen({ title, subtitle, showClock = false }: SimpleScreenProps) {
  return (
    <GameBackground>
      <div className="relative z-10 flex items-center justify-center min-h-screen px-4">
        <div className="text-center w-full max-w-md mx-auto">
          {showClock && (
            <div className="relative mb-6 sm:mb-8">
              <div className="relative w-16 h-16 sm:w-20 sm:h-20 bg-gradient-to-r from-blue-500/50 to-purple-500/50 border-2 border-white/30 rounded-full shadow-2xl flex items-center justify-center mx-auto animate-pulse">
                <Clock className="w-8 h-8 sm:w-10 sm:h-10 text-blue-200" />
              </div>
            </div>
          )}
          <h2 className="text-lg sm:text-xl font-bold text-white mb-2 break-words px-2">{title}</h2>
          <p className="text-xs sm:text-sm text-blue-200 break-words px-2">{subtitle}</p>
        </div>
      </div>
    </GameBackground>
  )
}
```
