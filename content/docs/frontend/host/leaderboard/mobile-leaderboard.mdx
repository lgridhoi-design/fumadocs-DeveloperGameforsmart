#### app/host/leaderboard/components/mobile-leaderboard.tsx

```tsx
import { RobustGoogleAvatar } from "@/components/player/robust-google-avatar"
import { PlayerName } from '@/components/player/player-name'
import { formatTime, type LeaderboardPlayer } from "./types"

interface MobileLeaderboardProps {
  players: LeaderboardPlayer[]
  getScoreStatusColor: (player: LeaderboardPlayer) => 'low' | 'high'
}

export function MobileLeaderboard({ players, getScoreStatusColor }: MobileLeaderboardProps) {
  return (
    <div className="md:hidden w-full max-w-md mx-auto pb-24 relative z-0 flex-1 flex flex-col min-h-0">
      {players.length > 0 && (
        <div className="mb-5">
          <div className="flex items-end justify-center gap-2">
            <PodiumCard rank={2} player={players[1]} avatarClass="from-slate-500 to-slate-600" nameClass="text-slate-200" blockClass="bg-gradient-to-b from-slate-500/70 to-slate-700/70 border border-slate-400/50" blockHeight="60px" numClass="text-2xl font-black text-slate-300" num="2" />

            <PodiumCard rank={1} player={players[0]} avatarClass="from-yellow-400 to-amber-500" nameClass="text-yellow-300" blockClass="bg-gradient-to-b from-yellow-400/80 to-amber-600/80 border border-yellow-300/60" blockHeight="90px" numClass="text-3xl font-black text-white drop-shadow" num="1" width={72} height={72} />

            <PodiumCard rank={3} player={players[2]} avatarClass="from-amber-500 to-orange-600" nameClass="text-amber-300" blockClass="bg-gradient-to-b from-amber-600/70 to-orange-700/70 border border-amber-500/50" blockHeight="45px" numClass="text-2xl font-black text-amber-200" num="3" />
          </div>
        </div>
      )}

      <div className="bg-blue-900/80 border border-blue-700/80 rounded-sm p-3 backdrop-blur-xl relative z-10 shadow-[0_30px_80px_rgba(15,23,42,0.45)] mt-4 flex-initial min-h-0 flex flex-col max-h-full">
        <div className="grid grid-cols-[minmax(36px,auto)_1.2fr_auto_auto] gap-3 items-center px-1 pb-3 mb-3 border-b border-blue-700/60 flex-shrink-0">
          <div className="text-[9px] uppercase tracking-[0.35em] text-yellow-400 font-bold text-left">PERINGKAT</div>
          <div className="text-[9px] uppercase tracking-[0.35em] text-yellow-400 font-bold text-left">NAMA</div>
          <div className="text-[9px] uppercase tracking-[0.35em] text-yellow-400 font-bold text-right">SKOR</div>
          <div className="text-[9px] uppercase tracking-[0.35em] text-yellow-400 font-bold text-right">WAKTU</div>
        </div>
        <div className="space-y-3 flex-1 min-h-0 overflow-y-auto pr-1">
          {players.map((player, index) => {
            const rank = index + 1
            const isLowScore = getScoreStatusColor(player) === 'low'
            const rankStyle = rank === 1 ? 'text-yellow-300' : rank === 2 ? 'text-slate-200' : rank === 3 ? 'text-amber-300' : 'text-slate-300'
            return (
              <div key={player.id} className="relative group">
                <div className="grid grid-cols-[minmax(36px,auto)_1.4fr_auto_auto] items-center gap-8">
                  <div className={`flex items-center justify-center text-[9px] font-bold ${rankStyle}`}>#{rank}</div>
                  <div className="flex items-center gap-3 min-w-0">
                    <div className="min-w-0">
                      <PlayerName name={player.nickname} className="text-[9px] font-semibold text-white" uppercase={false} />
                    </div>
                  </div>
                  <div className="text-right">
                    <div className={`${isLowScore ? 'text-red-500' : 'text-green-500'} font-bold text-[9px] `}>{Math.min(player.quizScore || 0, 100)}</div>
                  </div>
                  <div className="text-right flex flex-col items-end gap-0">
                    <span className={`${isLowScore ? 'text-red-500' : 'text-green-500'} font-semibold text-[9px] tracking-wide`}>{formatTime(player.timeSpent || 0)}</span>
                  </div>
                </div>
              </div>
            )
          })}
        </div>
      </div>
    </div>
  )
}

interface PodiumCardProps {
  rank: number
  player?: LeaderboardPlayer
  avatarClass: string
  nameClass: string
  blockClass: string
  blockHeight: string
  numClass: string
  num: string
  width?: number
  height?: number
}

function PodiumCard({
  rank,
  player,
  avatarClass,
  nameClass,
  blockClass,
  blockHeight,
  numClass,
  num,
  width,
  height,
}: PodiumCardProps) {
  if (!player) return <div className="flex-1" />
  const avatarCls = rank === 1 ? "w-18 h-18 rounded-full border-2 border-yellow-400 overflow-hidden shadow-xl mb-1.5 flex-shrink-0" : "w-14 h-14 rounded-full border-2 overflow-hidden shadow-lg mb-1.5 flex-shrink-0"
  const sizeStyle = rank === 1 ? { width: '4.5rem', height: '4.5rem' } : undefined
  return (
    <div className="flex flex-col items-center flex-1 min-w-0">
      <div className={avatarCls} style={sizeStyle}>
        {/^https?:\/\//.test(player.avatar) ? (
          <RobustGoogleAvatar avatarUrl={player.avatar} alt={player.nickname} width={width || 56} height={height || 56} className="w-full h-full object-cover" />
        ) : (
          <div className={`w-full h-full bg-gradient-to-br ${avatarClass} flex items-center justify-center`}>
            <span className="text-lg font-bold text-white">{player.nickname.charAt(0).toUpperCase()}</span>
          </div>
        )}
      </div>
      <div className="relative w-full text-center px-1">
        <PlayerName name={player.nickname} className={`${nameClass} text-xs font-bold block mx-auto truncate w-full`} uppercase={false} />
      </div>
      <div className={`w-full ${blockClass} rounded-t-lg mt-1.5 flex items-center justify-center`} style={{ height: blockHeight }}>
        <span className={numClass}>{num}</span>
      </div>
    </div>
  )
}
```
