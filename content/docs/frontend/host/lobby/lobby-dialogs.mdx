#### app/host/[roomCode]/lobby/components/lobby-dialogs.tsx

```tsx
"use client"

import { X, UserPlus, Search, AlertTriangle, CheckCircle, Lock } from "lucide-react"
import { Dialog, DialogContent, DialogTitle } from "@/components/ui/dialog"
import { Input } from "@/components/ui/input"
import { RobustGoogleAvatar } from "@/components/player/robust-google-avatar"
import { QRCodeSVG } from "qrcode.react"
import { ToastNotification } from "@/components/feedback/toast-notification"
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@/components/ui/tooltip"
import getDisplayName from '@/lib/display-name'
import { UsersIcon } from "lucide-react"

interface LobbyDialogsProps {
  t: (key: string, opts?: any) => string
  // Invite Friends
  inviteFriendsOpen: boolean
  setInviteFriendsOpen: (open: boolean) => void
  loadingFriends: boolean
  friendsList: any[]
  friendSearchTerm: string
  setFriendSearchTerm: (v: string) => void
  invitedFriends: string[]
  onInviteFriend: (friend: any) => void
  // Invite Groups
  inviteGroupsOpen: boolean
  setInviteGroupsOpen: (open: boolean) => void
  loadingGroups: boolean
  userGroups: any[]
  groupSearchTerm: string
  setGroupSearchTerm: (v: string) => void
  invitedGroups: string[]
  onInviteGroup: (group: any) => void
  // Leave
  showLeaveDialog: boolean
  setShowLeaveDialog: (open: boolean) => void
  isLeavingLobby: boolean
  onCancelLeave: () => void
  onConfirmLeave: () => void
  // Kick
  showKickDialog: boolean
  setShowKickDialog: (open: boolean) => void
  playerToKick: any
  onConfirmKick: () => void
  // Toast
  inviteToast: { open: boolean; type: "friend" | "group" | "error"; name: string; count?: number }
  onHideToast: () => void
  // QR overlay
  showQRModal: boolean
  setShowQRModal: (open: boolean) => void
  joinUrl: string
}

export function LobbyDialogs({
  t,
  inviteFriendsOpen,
  setInviteFriendsOpen,
  loadingFriends,
  friendsList,
  friendSearchTerm,
  setFriendSearchTerm,
  invitedFriends,
  onInviteFriend,
  inviteGroupsOpen,
  setInviteGroupsOpen,
  loadingGroups,
  userGroups,
  groupSearchTerm,
  setGroupSearchTerm,
  invitedGroups,
  onInviteGroup,
  showLeaveDialog,
  setShowLeaveDialog,
  isLeavingLobby,
  onCancelLeave,
  onConfirmLeave,
  showKickDialog,
  setShowKickDialog,
  playerToKick,
  onConfirmKick,
  inviteToast,
  onHideToast,
  showQRModal,
  setShowQRModal,
  joinUrl,
}: LobbyDialogsProps) {
  const truncateText = (text: string, maxLength: number = 20) => {
    if (!text) return ''
    if (text.length <= maxLength) return text
    return text.substring(0, maxLength) + '...'
  }

  const filteredFriends = friendsList.filter(f => {
    const name = getDisplayName({ nickname: f.nickname, fullname: f.fullname, username: f.username }) || t('friends.friend')
    return name.toLowerCase().includes(friendSearchTerm.toLowerCase())
  })

  const filteredGroups = userGroups.filter(g =>
    (g.userRole === 'owner' || g.userRole === 'admin') && g.name.toLowerCase().includes(groupSearchTerm.toLowerCase())
  )

  return (
    <>
      {/* Invite Friends Dialog */}
      <Dialog open={inviteFriendsOpen} onOpenChange={setInviteFriendsOpen}>
        <DialogContent className="w-[95vw] sm:w-[448px] bg-transparent border-none p-0 z-50 [&>button:last-child]:hidden mx-auto">
          <div className="w-full h-[480px] max-h-[85vh] flex flex-col relative bg-linear-to-br from-blue-500 to-cyan-600 rounded-lg border-4 border-black shadow-2xl p-3 sm:p-6">
            <button
              onClick={() => setInviteFriendsOpen(false)}
              className="absolute top-2 right-2 sm:top-3 sm:right-3 text-white/70 hover:text-white transition-colors z-10"
            >
              <X className="h-4 w-4 sm:h-5 sm:w-5" />
            </button>

            <div className="text-center mb-3 sm:mb-4 shrink-0">
              <div className="w-12 h-12 sm:w-16 sm:h-16 mx-auto bg-white border-2 border-black rounded flex items-center justify-center mb-2 sm:mb-3">
                <UserPlus className="h-6 w-6 sm:h-8 sm:w-8 text-blue-600" />
              </div>
              <DialogTitle className="text-lg sm:text-xl font-bold text-white mb-1 sm:mb-2 pixel-font">{t('friends.inviteFriends')}</DialogTitle>
              <p className="text-white/80 text-[10px] sm:text-xs pixel-font-sm px-2">{t('friends.inviteDesc')}</p>
            </div>

            <div className="mb-3 shrink-0">
              <div className="relative">
                <Input
                  type="text"
                  placeholder={t('common.search')}
                  value={friendSearchTerm}
                  onChange={(e) => setFriendSearchTerm(e.target.value)}
                  className="w-full bg-white/10 border border-white/30 rounded-lg px-3 py-2 text-white text-sm pixel-font-sm placeholder:text-white/40 focus:outline-none focus:border-white/60"
                />
                <Search className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-white/40" />
              </div>
            </div>

            <div className={`max-h-[40vh] sm:max-h-[400px] overflow-y-auto px-1 flex-1 min-h-0 ${loadingFriends ? 'flex flex-col items-center justify-center' : 'space-y-3'}`}>
              {loadingFriends ? (
                <div className="text-center py-8">
                  <div className="w-8 h-8 border-2 border-white border-t-transparent rounded-full animate-spin mx-auto"></div>
                  <p className="text-white/70 text-xs mt-2">{t('friends.loading')}</p>
                </div>
              ) : filteredFriends.length === 0 ? (
                <div className="text-center py-8">
                  <UserPlus className="h-12 w-12 text-white/40 mx-auto mb-2" />
                  <p className="text-white/60 text-xs">{friendSearchTerm ? t('friends.noFriendsMatch') : t('friends.noFriends')}</p>
                </div>
              ) : (
                filteredFriends.map((friend) => {
                  const isInvited = invitedFriends.includes(friend.id)
                  const friendName = getDisplayName({ nickname: friend.nickname, fullname: friend.fullname, username: friend.username }) || t('friends.friend')
                  return (
                    <div key={friend.id} className="bg-white/10 border border-white/30 rounded-lg p-2 sm:p-3 flex items-center gap-2 sm:gap-3">
                      <div className="w-8 h-8 sm:w-10 sm:h-10 rounded-full bg-gradient-to-br from-blue-400 to-cyan-400 flex items-center justify-center overflow-hidden shrink-0">
                        {friend.avatar_url ? (
                          <RobustGoogleAvatar avatarUrl={friend.avatar_url} alt={friendName} width={40} height={40} className="w-full h-full object-cover" />
                        ) : (
                          <span className="text-white font-bold text-xs sm:text-base">{friendName.charAt(0).toUpperCase()}</span>
                        )}
                      </div>
                      <div className="flex-1 min-w-0">
                        <TooltipProvider delayDuration={200}>
                          <Tooltip>
                            <TooltipTrigger asChild>
                              <p className="text-white font-bold text-xs sm:text-sm pixel-font truncate cursor-default max-w-[100px] min-[400px]:max-w-[140px] sm:max-w-none">{truncateText(friendName, 25)}</p>
                            </TooltipTrigger>
                            <TooltipContent className="bg-slate-900 border border-slate-700 text-white text-[10px] sm:text-xs z-[100]">
                              {friendName}
                            </TooltipContent>
                          </Tooltip>
                        </TooltipProvider>
                        <p className="text-white/60 text-[8px] sm:text-[9px] truncate">{friend.username || t('friends.online')}</p>
                      </div>
                      <button
                        onClick={() => onInviteFriend(friend)}
                        disabled={isInvited}
                        className={`px-2 py-1 sm:px-3 sm:py-1.5 rounded-lg border-2 border-black font-bold text-[9px] sm:text-[10px] transition-all hover:scale-105 shrink-0 ${isInvited ? 'bg-green-600 opacity-70' : 'bg-green-500 hover:bg-green-600'} text-white`}
                      >
                        {isInvited ? (
                          <span className="flex items-center gap-1">
                            <CheckCircle className="h-3 w-3" /> {t('friends.invited')}
                          </span>
                        ) : t('friends.invite')}
                      </button>
                    </div>
                  )
                })
              )}
            </div>
          </div>
        </DialogContent>
      </Dialog>

      {/* Invite Groups Dialog */}
      <Dialog open={inviteGroupsOpen} onOpenChange={setInviteGroupsOpen}>
        <DialogContent className="w-[95vw] sm:w-[448px] bg-transparent border-none p-0 z-50 [&>button:last-child]:hidden mx-auto">
          <div className="w-full h-[480px] max-h-[85vh] flex flex-col relative bg-linear-to-br from-blue-500 to-cyan-600 rounded-lg border-4 border-black shadow-2xl p-3 sm:p-6">
            <button
              onClick={() => setInviteGroupsOpen(false)}
              className="absolute top-2 right-2 sm:top-3 sm:right-3 text-white/70 hover:text-white transition-colors z-10"
            >
              <X className="h-4 w-4 sm:h-5 sm:w-5" />
            </button>

            <div className="text-center mb-3 sm:mb-4 shrink-0">
              <div className="w-12 h-12 sm:w-16 sm:h-16 mx-auto bg-white border-2 border-black rounded flex items-center justify-center mb-2 sm:mb-3">
                <svg
                  xmlns="http://www.w3.org/2000/svg"
                  width="28"
                  height="24"
                  viewBox="0 0 28 24"
                  fill="none"
                  stroke="currentColor"
                  strokeWidth="2"
                  strokeLinecap="round"
                  strokeLinejoin="round"
                  className="h-6 w-6 sm:h-8 sm:w-8 text-blue-600"
                >
                  <path d="M16 21v-2a4 4 0 0 0-4-4H6a4 4 0 0 0-4 4v2" />
                  <path d="M16 3.128a4 4 0 0 1 0 7.744" />
                  <path d="M22 21v-2a4 4 0 0 0-3-3.87" />
                  <circle cx="9" cy="7" r="4" />
                  <line x1="24" y1="8" x2="24" y2="14" />
                  <line x1="21" y1="11" x2="27" y2="11" />
                </svg>
              </div>
              <DialogTitle className="text-lg sm:text-xl font-bold text-white mb-1 sm:mb-2 pixel-font">{t('groups.inviteGroups')}</DialogTitle>
              <p className="text-white/80 text-[10px] sm:text-xs pixel-font-sm px-2">{t('groups.inviteDesc')}</p>
            </div>

            <div className="mb-3 shrink-0">
              <div className="relative">
                <Input
                  type="text"
                  placeholder={t('common.search')}
                  value={groupSearchTerm}
                  onChange={(e) => setGroupSearchTerm(e.target.value)}
                  className="w-full bg-white/10 border border-white/30 rounded-lg px-3 py-2 text-white text-sm pixel-font-sm placeholder:text-white/40 focus:outline-none focus:border-white/60"
                />
                <Search className="absolute right-3 top-1/2 -translate-y-1/2 h-4 w-4 text-white/40" />
              </div>
            </div>

            <div className={`max-h-[40vh] sm:max-h-[400px] overflow-y-auto px-1 flex-1 min-h-0 ${loadingGroups ? 'flex flex-col items-center justify-center' : 'space-y-3'}`}>
              {loadingGroups ? (
                <div className="text-center py-8">
                  <div className="w-8 h-8 border-2 border-white border-t-transparent rounded-full animate-spin mx-auto"></div>
                  <p className="text-white/70 text-xs mt-2">{t('groups.loading')}</p>
                </div>
              ) : filteredGroups.length === 0 ? (
                <div className="text-center py-8">
                  <UsersIcon className="h-12 w-12 text-white/40 mx-auto mb-2" />
                  <p className="text-white/60 text-xs">{groupSearchTerm ? t('groups.noGroupsMatch') : t('groups.noGroups')}</p>
                </div>
              ) : (
                filteredGroups.map((group) => {
                  const isInvited = invitedGroups.includes(group.id)
                  const canInvite = group.userRole === 'owner' || group.userRole === 'admin'

                  return (
                    <div key={group.id} className="bg-white/10 border border-white/30 rounded-lg p-2 sm:p-3 flex items-center justify-between gap-2">
                      <div className="flex-1 min-w-0">
                        <TooltipProvider delayDuration={200}>
                          <Tooltip>
                            <TooltipTrigger asChild>
                              <p className="text-white font-bold text-xs sm:text-sm pixel-font truncate cursor-default max-w-[100px] min-[400px]:max-w-[140px] sm:max-w-none">{truncateText(group.name, 25)}</p>
                            </TooltipTrigger>
                            <TooltipContent className="bg-slate-900 border border-slate-700 text-white text-[10px] sm:text-xs z-[100]">
                              {group.name}
                            </TooltipContent>
                          </Tooltip>
                        </TooltipProvider>
                        <p className="text-white/60 text-[8px] sm:text-[10px] flex items-center gap-1 mt-0.5 sm:mt-1">
                          <UsersIcon className="h-2.5 w-2.5 sm:h-3 sm:w-3" />
                          <span>{group.members_count} {t('groups.members')}</span>
                        </p>
                      </div>

                      {canInvite ? (
                        <button
                          onClick={() => onInviteGroup(group)}
                          disabled={isInvited}
                          className={`px-2 py-1 sm:px-3 sm:py-1.5 rounded-lg border-2 border-black font-bold text-[9px] sm:text-[10px] transition-all hover:scale-105 shrink-0 ${isInvited ? 'bg-green-600 opacity-70' : 'bg-green-500 hover:bg-green-600'} text-white`}
                        >
                          {isInvited ? (
                            <span className="flex items-center gap-1">
                              <CheckCircle className="h-3 w-3" /> {t('groups.invited')}
                            </span>
                          ) : t('groups.invite')}
                        </button>
                      ) : (
                        <TooltipProvider delayDuration={200}>
                          <Tooltip>
                            <TooltipTrigger asChild>
                              <div className="px-3 py-1.5 rounded-lg border-2 border-gray-500 bg-gray-600/50 text-gray-400 font-bold text-[10px] cursor-not-allowed shrink-0 flex items-center gap-1">
                                <Lock className="h-3 w-3" />
                                <span>{t('groups.invite')}</span>
                              </div>
                            </TooltipTrigger>
                            <TooltipContent
                              side="left"
                              className="bg-gray-900 border border-gray-600 text-white text-[10px] py-2 px-3"
                            >
                              {t('groups.memberCannotInvite')}
                            </TooltipContent>
                          </Tooltip>
                        </TooltipProvider>
                      )}
                    </div>
                  )
                })
              )}
            </div>
          </div>
        </DialogContent>
      </Dialog>

      {/* Leave Dialog */}
      <Dialog open={showLeaveDialog} onOpenChange={(open) => { if (!isLeavingLobby) setShowLeaveDialog(open) }}>
        <DialogContent className="w-[95vw] max-w-md bg-transparent border-none p-0 [&>button:last-child]:hidden mx-auto">
          <div className="relative bg-linear-to-br from-blue-500 to-purple-600 rounded-lg border-4 border-black shadow-2xl p-4 sm:p-6 text-center">
            <AlertTriangle className="h-12 w-12 text-yellow-400 mx-auto mb-4" />
            <DialogTitle className="text-xl font-bold text-white mb-2">{t('lobby.leaveWarning')}</DialogTitle>
            <p className="text-white/80 text-sm mb-6">{t('lobby.leaveWarningDesc')}</p>
            <div className="flex gap-4 justify-center">
              <button
                onClick={onCancelLeave}
                className="bg-gray-500 hover:bg-gray-600 border-2 border-black rounded-lg text-white px-6 py-2 font-bold"
              >
                {t("common.cancel")}
              </button>
              <button
                onClick={onConfirmLeave}
                disabled={isLeavingLobby}
                className={`bg-red-500 hover:bg-red-600 border-2 border-black rounded-lg text-white px-6 py-2 font-bold disabled:opacity-50 disabled:cursor-not-allowed disabled:pointer-events-none ${isLeavingLobby ? 'opacity-50 cursor-not-allowed pointer-events-none' : ''}`}
              >
                {isLeavingLobby ? "..." : t("common.leave")}
              </button>
            </div>
          </div>
        </DialogContent>
      </Dialog>

      {/* Kick Dialog */}
      <Dialog open={showKickDialog} onOpenChange={setShowKickDialog}>
        <DialogContent className="w-[95vw] max-w-md bg-transparent border-none p-0 [&>button:last-child]:hidden mx-auto">
          <div className="relative bg-linear-to-br from-red-500 to-red-600 rounded-lg border-4 border-black shadow-2xl p-4 sm:p-6 text-center">
            <AlertTriangle className="h-12 w-12 text-yellow-400 mx-auto mb-4" />
            <DialogTitle className="text-xl font-bold text-white mb-2">{t('lobby.kickPlayer')}</DialogTitle>
            <p className="text-white/80 text-sm mb-6 px-2 break-words">
              {t('lobby.confirmKick')} <span className="font-bold break-all">{playerToKick?.nickname?.toUpperCase()}</span>?
            </p>
            <div className="flex gap-4 justify-center">
              <button
                onClick={() => setShowKickDialog(false)}
                className="bg-gray-500 hover:bg-gray-600 border-2 border-black rounded-lg text-white px-6 py-2 font-bold"
              >
                {t('common.cancel')}
              </button>
              <button
                onClick={onConfirmKick}
                className="bg-red-700 hover:bg-red-800 border-2 border-black rounded-lg text-white px-6 py-2 font-bold"
              >
                {t('common.kick')}
              </button>
            </div>
          </div>
        </DialogContent>
      </Dialog>

      <ToastNotification
        isOpen={inviteToast.open}
        onClose={onHideToast}
        type={inviteToast.type}
        name={inviteToast.name}
        count={inviteToast.count}
        duration={3000}
      />

      {showQRModal && (
        <div
          className="fixed inset-0 bg-black z-[100] flex flex-col justify-center items-center p-4 cursor-pointer select-none animate-in fade-in duration-200"
          onClick={() => setShowQRModal(false)}
        >
          <div
            className="bg-white p-3 sm:p-4 rounded-2xl max-w-[90vw] max-h-[90vh] flex items-center justify-center shadow-2xl animate-in zoom-in-95 duration-200"
            onClick={(e) => e.stopPropagation()}
          >
            <QRCodeSVG
              value={joinUrl}
              size={typeof window !== 'undefined' ? Math.min(window.innerWidth * 0.85, window.innerHeight * 0.85, 750) : 750}
              level="H"
              includeMargin={false}
            />
          </div>
        </div>
      )}
    </>
  )
}
```
