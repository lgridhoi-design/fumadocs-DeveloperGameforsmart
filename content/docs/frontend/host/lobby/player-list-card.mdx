#### app/host/[roomCode]/lobby/components/player-list-card.tsx

```tsx
"use client"

import { Users, UserPlus, LogOut, Play, X, ChevronLeft, ChevronRight } from "lucide-react"
import { Button } from "@/components/ui/button"
import { RobustGoogleAvatar } from "@/components/player/robust-google-avatar"
import { PlayerName } from "@/components/player/player-name"
import type { Room } from "@/lib/room-manager"

interface PlayerListCardProps {
  t: (key: string, opts?: any) => string
  currentRoom: Room | null
  currentPlayers: any[]
  hostId: string | null
  roomCode: string | null
  quizId: string | null
  gameStarted: boolean
  playerCountChanged: boolean
  playerLeft: boolean
  totalPages: number
  currentPage: number
  onInviteGroups: () => void
  onInviteFriends: () => void
  onExit: () => void
  onStartGame: () => void
  onKickPlayer: (player: any) => void
  onNextPage: () => void
  onPrevPage: () => void
}

export function PlayerListCard({
  t,
  currentRoom,
  currentPlayers,
  hostId,
  roomCode,
  quizId,
  gameStarted,
  playerCountChanged,
  playerLeft,
  totalPages,
  currentPage,
  onInviteGroups,
  onInviteFriends,
  onExit,
  onStartGame,
  onKickPlayer,
  onNextPage,
  onPrevPage,
}: PlayerListCardProps) {
  return (
    <div className="relative pixel-lobby-container">
      <div className="absolute inset-0 bg-linear-to-br from-green-600 to-cyan-600 rounded-lg transform rotate-1 pixel-button-shadow"></div>
      <div className="relative bg-linear-to-br from-green-500 to-cyan-500 rounded-lg border-2 sm:border-4 border-black shadow-2xl pixel-lobby-card">
        <div className="p-4 sm:p-6">
          <div className="flex flex-row items-center justify-between mb-4 gap-2">
            <div className="flex flex-row items-center justify-start shrink-0">
              <div className={`inline-flex items-center justify-center h-10 bg-white border-2 border-black rounded-lg px-3 sm:px-4 transition-all duration-500 ${playerCountChanged ? 'animate-pulse bg-green-100 border-green-400' : playerLeft ? 'animate-pulse bg-red-100 border-red-400' : ''}`}>
                <span className="text-black font-bold text-xs pixel-font flex items-center gap-1.5">
                  {playerCountChanged ? '🎉' : playerLeft ? '👋' : ''}
                  <Users className="h-4 w-4 shrink-0" />
                  <span>{currentRoom?.players.length || 0}</span>
                </span>
              </div>
            </div>

            <div className="flex flex-row items-center gap-2 sm:gap-3 justify-end">
              {currentRoom && !gameStarted && (
                <>
                  <div className="relative pixel-button-container">
                    <div className="absolute inset-0 bg-linear-to-br from-blue-600 to-cyan-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
                    <button
                      onClick={onInviteGroups}
                      className="relative bg-linear-to-br from-blue-500 to-cyan-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-blue-400 hover:to-cyan-500 transform hover:scale-105 transition-all duration-200 font-bold w-10 h-10 flex items-center justify-center"
                      title={t('groups.inviteGroups')}
                    >
                      <svg
                        xmlns="http://www.w3.org/2000/svg"
                        width="28"
                        height="24"
                        viewBox="0 0 28 24"
                        fill="none"
                        stroke="currentColor"
                        strokeWidth="2"
                        strokeLinecap="round"
                        strokeLinejoin="round"
                        className="lucide lucide-users-plus h-5 w-5"
                      >
                        <path d="M16 21v-2a4 4 0 0 0-4-4H6a4 4 0 0 0-4 4v2" />
                        <path d="M16 3.128a4 4 0 0 1 0 7.744" />
                        <path d="M22 21v-2a4 4 0 0 0-3-3.87" />
                        <circle cx="9" cy="7" r="4" />
                        <line x1="24" y1="8" x2="24" y2="14" />
                        <line x1="21" y1="11" x2="27" y2="11" />
                      </svg>
                    </button>
                  </div>

                  <div className="relative pixel-button-container">
                    <div className="absolute inset-0 bg-linear-to-br from-blue-600 to-blue-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
                    <button
                      onClick={onInviteFriends}
                      className="relative bg-linear-to-br from-blue-500 to-blue-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-blue-400 hover:to-blue-500 transform hover:scale-105 transition-all duration-200 font-bold w-10 h-10 flex items-center justify-center"
                      title={t('friends.inviteFriends')}
                    >
                      <UserPlus className="h-4 w-4" />
                    </button>
                  </div>
                </>
              )}

              {currentRoom && !gameStarted && (
                <div className="relative pixel-button-container flex-initial">
                  <div className="absolute inset-0 bg-linear-to-br from-red-600 to-red-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
                  <button
                    onClick={onExit}
                    className="relative bg-linear-to-br from-red-500 to-red-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-red-400 hover:to-red-500 transform hover:scale-105 transition-all duration-200 font-bold px-0 md:px-5 w-10 h-10 md:w-auto md:h-auto md:py-3 flex items-center justify-center gap-1.5 md:gap-2.5 animate-pulse"
                    style={{ animationDuration: '3s' }}
                  >
                    <LogOut className="h-5 w-5 md:h-4 md:w-4 shrink-0" />
                    <span className="pixel-font text-xs whitespace-nowrap hidden md:inline">{t('lobby.exitRoom')}</span>
                  </button>
                </div>
              )}

              {currentRoom && !gameStarted && (
                <div className="relative pixel-button-container hidden md:block flex-initial">
                  <div className="absolute inset-0 bg-linear-to-br from-purple-600 to-purple-700 rounded-lg transform rotate-1 pixel-button-shadow"></div>
                  <button
                    onClick={onStartGame}
                    disabled={!roomCode || !hostId || !currentRoom || currentRoom.players.length === 0}
                    className="relative bg-linear-to-br from-purple-500 to-purple-600 border-2 border-black rounded-lg text-white hover:bg-linear-to-br hover:from-purple-400 hover:to-purple-500 transform hover:scale-105 transition-all duration-200 font-bold px-5 py-3 flex items-center justify-center gap-2.5 disabled:opacity-50 disabled:cursor-not-allowed w-full text-center"
                  >
                    <Play className="h-4 w-4 shrink-0" />
                    <span className="pixel-font text-xs whitespace-nowrap">{t('lobby.startGame')}</span>
                  </button>
                </div>
              )}

              {gameStarted && (
                <div className="bg-yellow-400 border-2 border-black rounded px-4 py-2.5 w-full sm:w-auto text-center">
                  <span className="text-black font-bold text-xs pixel-font">{t('lobby.gameStarted')}</span>
                </div>
              )}
            </div>
          </div>

          {currentRoom && currentRoom.players.length === 0 ? (
            <div className="text-center py-6 sm:py-8">
              <div className="w-12 h-12 sm:w-16 sm:h-16 mx-auto bg-white border-2 border-black rounded flex items-center justify-center mb-4">
                <Users className="h-6 w-6 sm:h-8 sm:w-8 text-black" />
              </div>
              <div className="bg-white border-2 border-black rounded px-3 sm:px-4 py-2 inline-block">
                <p className="text-black font-bold text-[10px] sm:text-xs pixel-font mb-2">{t('lobby.noPlayersYet')}</p>
                <p className="text-black text-[8px] sm:text-[9px] pixel-font opacity-70 leading-none">{t('lobby.shareRoomCode')}</p>
              </div>
            </div>
          ) : (
            <div className="space-y-4">
              <div className="grid grid-cols-2 sm:grid-cols-4 gap-2">
                {currentPlayers.map((player: any) => (
                  <div key={player.id} className="bg-white border-2 border-black rounded p-2 relative group">
                    {hostId && (
                      <button
                        onClick={(e) => { e.stopPropagation(); onKickPlayer(player) }}
                        className="absolute top-1 right-1 lg:-top-1 lg:-right-1 flex items-center justify-center z-10 p-1 min-w-0 min-h-0 bg-transparent border-transparent lg:p-0 lg:w-5 lg:h-5 lg:bg-red-500 lg:border-solid lg:border lg:border-black lg:rounded-full lg:hover:bg-red-600 lg:opacity-0 lg:group-hover:opacity-100 transition-all duration-150"
                      >
                        <X className="h-4 w-4 text-black drop-shadow-md lg:drop-shadow-none lg:h-3 lg:w-3 lg:text-white" />
                      </button>
                    )}
                    <div className="flex flex-col items-center gap-1 w-full">
                      <RobustGoogleAvatar avatarUrl={player.avatar} alt={player.nickname} width={40} height={40} className="w-10 h-10 rounded-full flex-shrink-0" />
                      <div className="relative w-full">
                        <PlayerName name={player.nickname} className="font-bold text-black pixel-font text-[8px] text-center block w-full px-1 truncate" uppercase={true} />
                      </div>
                    </div>
                  </div>
                ))}
              </div>

              {totalPages > 1 && (
                <div className="flex items-center justify-between">
                  <Button onClick={onPrevPage} disabled={currentPage === 0} className="bg-gray-500 border-2 border-black rounded-lg text-white px-3 py-1 disabled:opacity-50">
                    <ChevronLeft className="h-3 w-3" />
                  </Button>
                  <span className="text-white text-xs pixel-font">{currentPage + 1} / {totalPages}</span>
                  <Button onClick={onNextPage} disabled={currentPage === totalPages - 1} className="bg-gray-500 border-2 border-black rounded-lg text-white px-3 py-1 disabled:opacity-50">
                    <ChevronRight className="h-3 w-3" />
                  </Button>
                </div>
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
```
