```tsx
"use client"

import { useState, useEffect } from "react"
import { X, Globe, BookOpen, Heart, Gamepad2 } from "lucide-react"
import { Button } from "@/components/ui/button"
import {
  Dialog,
  DialogContent,
} from "@/components/ui/dialog"
import { getLanguageDisplayName, translateCategoryFn } from "./categories"

interface QuizDetailDialogProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  quiz: any
  onStart: () => void
  onTryOut: () => void
  getCategoryColor: (category: string) => string
  favoriteCount?: number
  isLoading?: boolean
}

export function QuizDetailDialog({
  open,
  onOpenChange,
  quiz,
  onStart,
  onTryOut,
  getCategoryColor,
  favoriteCount = 0,
  isLoading = false,
}: QuizDetailDialogProps) {
  const [showContent, setShowContent] = useState(false)
  const [isDescExpanded, setIsDescExpanded] = useState(false)

  useEffect(() => {
    if (!isLoading && quiz) {
      requestAnimationFrame(() => {
        setShowContent(true)
      })
    } else {
      setShowContent(false)
    }
  }, [isLoading, quiz])

  useEffect(() => {
    if (!open) {
      setShowContent(false)
      setIsDescExpanded(false)
    }
  }, [open])

  if (!quiz && !isLoading) return null

  const totalQuestions = quiz?.question_count || quiz?.questions?.length || 0
  const timesPlayed = quiz?.played || quiz?.times_played || 0
  const language = getLanguageDisplayName(quiz?.language)
  const description = quiz?.description || "Tidak ada deskripsi tersedia untuk kuis ini."

  const shouldTruncate = description.length > 120
  const displayedDescription = shouldTruncate && !isDescExpanded
    ? `${description.slice(0, 120)}...`
    : description

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent
        className="bg-gradient-to-br from-blue-500 to-purple-500 border-4 border-black shadow-2xl max-w-sm sm:max-w-md w-[92%] rounded-2xl p-0 overflow-hidden [&>button:last-child]:hidden fixed top-[50%] left-[50%] translate-x-[-50%] translate-y-[-50%] z-50 rtl:left-auto rtl:right-[50%] rtl:translate-x-[50%]"
      >
        {isLoading ? (
          <div className="p-6 min-h-80 flex items-center justify-center">
            <div className="flex flex-col items-center justify-center">
              <div className="relative">
                <div className="w-12 h-12 border-3 border-white/20 border-t-white rounded-full animate-spin"></div>
                <div className="absolute inset-0 flex items-center justify-center">
                  <div className="w-4 h-4 bg-white rounded-sm animate-pulse"></div>
                </div>
              </div>
            </div>
          </div>
        ) : (
          <div
            className={`p-6 transition-all duration-300 ease-out ${showContent
              ? 'scale-100 opacity-100'
              : 'scale-95 opacity-0'
              }`}
          >
            <button
              onClick={() => onOpenChange(false)}
              className="absolute top-4 right-4 rtl:right-auto rtl:left-4 text-white/70 hover:text-white transition-colors p-1"
            >
              <X className="w-5 h-5" />
            </button>

            <div className="flex items-center flex-wrap gap-2 mb-4 pr-8 rtl:pr-0 rtl:pl-8">
              <span className={`px-3 py-1 rounded-full text-[10px] sm:text-xs font-black border-2 border-black text-white uppercase pixel-font-sm shadow-md ${getCategoryColor(quiz?.category || 'general')} flex items-center justify-center`}>
                <span>
                  {translateCategoryFn(quiz?.category)}
                </span>
              </span>
              <span className="px-3 py-1 rounded-full text-[10px] sm:text-xs font-black bg-black/20 text-cyan-300 border-2 border-black flex items-center gap-1.5 shadow-md">
                <Globe className="w-3.5 h-3.5 text-cyan-300" />
                <span>
                  {language}
                </span>
              </span>
            </div>

            <div className="space-y-4">
              <div className="space-y-2">
                <h3 className="text-white text-xl sm:text-2xl font-black pixel-font leading-tight wrap-break-word uppercase mt-3 drop-shadow-[0_2px_2px_rgba(0,0,0,0.6)]">
                  <span>
                    {quiz?.title}
                  </span>
                </h3>
                <p className="text-white/50 text-xs sm:text-sm leading-relaxed font-semibold drop-shadow-[0_1px_1px_rgba(0,0,0,0.4)]">
                  <span>
                    {displayedDescription}
                    {shouldTruncate && (
                      <button
                        onClick={() => setIsDescExpanded(!isDescExpanded)}
                        className="text-cyan-300 font-extrabold hover:text-cyan-200 transition-colors ml-1 inline-block focus:outline-none"
                      >
                        {isDescExpanded ? "Lebih sedikit" : "Selengkapnya"}
                      </button>
                    )}
                  </span>
                </p>
              </div>

              <div className="grid grid-cols-3 gap-2 py-4 border-2 border-black bg-black/20 rounded-xl my-4 text-center shadow-[4px_4px_0px_rgba(0,0,0,0.4)]">
                <div className="flex flex-col items-center justify-center">
                  <BookOpen className="w-5 h-5 text-cyan-300 mb-1 drop-shadow-[0_1.5px_0_rgba(0,0,0,0.5)]" />
                  <span className="text-white text-base sm:text-lg font-bold pixel-font leading-none drop-shadow-[0_1.5px_0_rgba(0,0,0,0.6)]">{totalQuestions}</span>
                  <span className="text-white/80 text-[7px] sm:text-[8px] font-bold tracking-wider mt-1 uppercase leading-none">
                    SOAL
                  </span>
                </div>

                <div className="flex flex-col items-center justify-center border-l-2 border-black border-r-2">
                  <Gamepad2 className="w-5 h-5 text-green-300 mb-1 drop-shadow-[0_1.5px_0_rgba(0,0,0,0.5)]" />
                  <span className="text-white text-base sm:text-lg font-bold pixel-font leading-none drop-shadow-[0_1.5px_0_rgba(0,0,0,0.6)]">{timesPlayed}x</span>
                  <span className="text-white/80 text-[7px] sm:text-[8px] font-bold tracking-wider mt-1 uppercase leading-none">
                    DIMAINSKAN
                  </span>
                </div>

                <div className="flex flex-col items-center justify-center">
                  <Heart className="w-5 h-5 text-pink-400 fill-current mb-1 drop-shadow-[0_1.5px_0_rgba(0,0,0,0.5)]" />
                  <span className="text-white text-base sm:text-lg font-bold pixel-font leading-none drop-shadow-[0_1.5px_0_rgba(0,0,0,0.6)]">{favoriteCount}</span>
                  <span className="text-white/80 text-[7px] sm:text-[8px] font-bold tracking-wider mt-1 uppercase leading-none">
                    FAVORIT
                  </span>
                </div>
              </div>
            </div>

            <div className="pt-2 flex gap-2">
              <Button
                onClick={onTryOut}
                className="flex-1 bg-cyan-500 hover:bg-cyan-600 border-4 border-black rounded-xl text-white font-extrabold text-xs sm:text-sm py-5 sm:py-6 shadow-[4px_4px_0px_rgba(0,0,0,1)] flex items-center justify-center gap-1 transform active:translate-x-[2px] active:translate-y-[2px] active:shadow-none transition-all duration-150"
              >
                <span className="pixel-font font-black uppercase tracking-wider">
                  LATIHAN
                </span>
              </Button>
              <Button
                onClick={onStart}
                className={`flex-1 ${getCategoryColor(quiz?.category || 'general')} hover:brightness-110 border-4 border-black rounded-xl text-white font-extrabold text-xs sm:text-sm py-5 sm:py-6 shadow-[4px_4px_0px_rgba(0,0,0,1)] flex items-center justify-center gap-1 transform active:translate-x-[2px] active:translate-y-[2px] active:shadow-none transition-all duration-150`}
              >
                <span className="pixel-font font-black uppercase tracking-wider">
                  MULAI KUIS
                </span>
              </Button>
            </div>
          </div>
        )}
      </DialogContent>
    </Dialog>
  )
}
```
