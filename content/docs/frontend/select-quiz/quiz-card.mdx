#### app/select-quiz/components/quiz-card.tsx

```tsx
import { HelpCircle, Play, Heart } from "lucide-react"
import { CachedImage } from "@/components/layout/cached-image"
import {
  Tooltip,
  TooltipContent,
  TooltipProvider,
  TooltipTrigger,
} from "@/components/ui/tooltip"
import { translateCategoryFn } from "./categories"

interface QuizCardProps {
  quiz: any
  isFavorite: boolean
  onToggleFavorite: (quizId: string, e: React.MouseEvent) => void
  onQuizClick: (quiz: any) => void
  getCategoryColor: (category: string) => string
  getQuizBackgroundImage: (quiz: any) => string
}

export function QuizCard({ quiz, isFavorite, onToggleFavorite, onQuizClick, getCategoryColor, getQuizBackgroundImage }: QuizCardProps) {
  return (
    <div
      onClick={() => onQuizClick(quiz)}
      className="group relative mx-auto w-full sm:max-w-[200px] bg-white border-2 border-black transition-all duration-300 hover:-translate-y-1 hover:shadow-[8px_8px_0px_0px_rgba(0,0,0,1)] cursor-pointer overflow-hidden h-[225px] sm:aspect-[3/4] sm:h-auto"
    >
      <div className="absolute inset-0">
        <CachedImage
          src={getQuizBackgroundImage(quiz)}
          alt={quiz.title}
          className="w-full h-full object-cover transition-transform duration-500 group-hover:scale-110 grayscale-[0.2] group-hover:grayscale-0"
        />
      </div>
      <div className="absolute inset-0 bg-gradient-to-t from-black via-black/40 to-transparent"></div>

      <div className="absolute top-1.5 left-1.5 right-10 rtl:right-1.5 rtl:left-auto z-10 flex items-start">
        <div className={`w-fit max-w-[calc(100%-0.25rem)] rounded-full border border-white/20 bg-black/30 px-2 py-0.5 pr-2.5 pl-2.5 rtl:pl-2.5 rtl:pr-2.5 backdrop-blur-md shadow-md ${getCategoryColor(quiz.category)}`}>
          <div className="flex w-full min-w-0 items-center gap-1">
            <div className="h-0.5 w-0.5 shrink-0 rounded-full bg-white animate-pulse"></div>
            <span className="min-w-0 max-w-full text-[5.5px] font-bold uppercase tracking-wider text-white pixel-font leading-tight whitespace-nowrap rtl:text-right">
              {translateCategoryFn(quiz.category)}
            </span>
          </div>
        </div>
      </div>

      <div onClick={(e) => onToggleFavorite(quiz.id, e)} className="absolute top-1.5 right-1.5 rtl:left-1.5 rtl:right-auto z-20 p-1 rounded-full bg-black/40 backdrop-blur-md border border-white/20 text-white hover:bg-pink-500 transition-colors shadow-lg group/heart">
        <Heart
          className={`h-4 w-4 transition-transform group-hover/heart:scale-125 ${isFavorite ? 'text-white' : ''}`}
          fill={isFavorite ? "currentColor" : "none"}
          strokeWidth={2.5}
        />
      </div>

      <div className="absolute bottom-0 left-0 right-0 p-2 sm:p-2.5 text-white">
        <TooltipProvider delayDuration={300}>
          <Tooltip>
            <TooltipTrigger asChild>
              <h3 className="text-[11px] sm:text-xs font-bold leading-tight mb-2 group-hover:text-blue-300 transition-colors line-clamp-2 cursor-help">{quiz.title}</h3>
            </TooltipTrigger>
            <TooltipContent
              side="top"
              className="bg-slate-950 text-white border-2 border-white rounded-none shadow-[6px_6px_0px_rgba(0,0,0,1)] max-w-[280px] text-[10px] pixel-font p-3 animate-in zoom-in-90 duration-200"
            >
              {quiz.title}
            </TooltipContent>
          </Tooltip>
        </TooltipProvider>

        <div className="flex items-center justify-between gap-1 mt-1">
          <div className="flex items-center gap-1 text-[6px] sm:text-[7px] pixel-font text-white/90 transition-colors bg-white/10 px-1 py-0.5 rounded border border-white/10 backdrop-blur-sm shrink-0 uppercase">
            <HelpCircle className="h-2 w-2" />
            <span>{quiz.question_count || quiz.questions?.length || 0}</span>
          </div>
          <div className="flex items-center gap-1 text-[6px] sm:text-[7px] pixel-font text-white/90 bg-white/10 px-1 py-0.5 rounded border border-white/10 backdrop-blur-sm shrink-0 uppercase">
            <Play className="h-2 w-2 fill-current" />
            <span>{quiz.played || 0}</span>
          </div>
        </div>
      </div>
    </div>
  )
}

export function QuizCardSkeleton() {
  return (
    <div className="mx-auto w-full sm:max-w-[200px] bg-white/5 border-2 border-black/20 overflow-hidden h-[225px] sm:aspect-[3/4] sm:h-auto p-2.5 flex flex-col justify-end gap-2 animate-pulse">
      <div className="w-full h-4 bg-white/20 mb-1"></div>
      <div className="w-2/3 h-4 bg-white/20 mb-3"></div>
      <div className="flex gap-1.5">
        <div className="w-14 h-3 bg-white/10"></div>
        <div className="w-14 h-3 bg-white/10"></div>
      </div>
    </div>
  )
}
```
